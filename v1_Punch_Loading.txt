FUNCTION_BLOCK Punch_Loading
VAR_INPUT
  LoadStartCmd      : BOOL;   (* State_Machine'den gelen başlat komutu *)
  Reset             : BOOL;   (* SOFT reset: state/substate resetlemez, sadece requestleri keser *)
  Abort_Cycle       : BOOL;   (* HARD abort: tüm FB içi state/substate/flagleri resetler *)
  Automatic_Page    : BOOL;   (* TRUE: otomatik sayfa aktif, FALSE: otomatik dışı -> pause zorunlu *)
END_VAR

VAR_IN_OUT
  General_Stop      : BOOL;
  General_Start     : BOOL;   (* Enable/pause: FALSE iken pause, TRUE iken devam *)
END_VAR

VAR_OUTPUT
  LoadStartAck      : BOOL;   (* State_Machine'e ACK *)
  LoadBusy          : BOOL;   (* FB aktif çalışıyor *)
  LoadError         : BOOL;   (* hata flag *)
  LoadDone          : BOOL;   (* tüm load bitti *)
END_VAR

VAR RETAIN
  {attribute 'ac_persist':='PersistenceChannel'} Local_Shuttle_State      : INT := 0;   (* Shuttle state *)
  {attribute 'ac_persist':='PersistenceChannel'} Local_WareHouse_State    : INT := 0;   (* Warehouse state *)
  {attribute 'ac_persist':='PersistenceChannel'} Local_RotArm_State       : INT := 0;   (* Rotation Arm state *)

  {attribute 'ac_persist':='PersistenceChannel'} LoadedPunchCount         : INT := 0;

  {attribute 'ac_persist':='PersistenceChannel'} PLoad_WH_SubStates       : INT := 0;
  {attribute 'ac_persist':='PersistenceChannel'} PLoad_SH_SubStates       : INT := 0;
  {attribute 'ac_persist':='PersistenceChannel'} PLoad_RotArm_SubStates   : INT := 0;

  lastStartCmd             : BOOL := FALSE;
  initiated                : BOOL := FALSE;

  Current_PunchIndex       : INT := 0;      (* 0 = yok, 1..15 geçerli *)
  Current_PunchLoaded      : BOOL := FALSE;
  LastInitializedPunch     : INT := 0;

  (* Warehouse internal flags *)
  WH_TargetDrawerIdx       : INT := 0;      (* hedef drawer index 1..16 *)
  WH_TargetRackID          : INT := 0;      (* hedef rack ID 1..4 *)
  WH_NeedRackBring         : BOOL := FALSE;
  WH_NeedRackChange        : BOOL := FALSE;
  WH_SameRack              : BOOL := FALSE;

  WH_R_MoveRequested       : BOOL := FALSE;
  WH_X_MoveRequested       : BOOL := FALSE;
  WH_R_Done                : BOOL := FALSE;
  WH_X_Done                : BOOL := FALSE;
  WH_WaitRackInPos         : BOOL := FALSE;

  WH_Eps                   : REAL := 0.01;

  WH_R_LoadRequested       : BOOL := FALSE;
  WH_R_LoadDone            : BOOL := FALSE;
  R_Not_Safety_Zone        : BOOL := FALSE;
  Rack_Presence_Error      : BOOL := FALSE;

  (* Shuttle placeholders - sonraki aşamalar için *)
  SH_DX_MoveRequested      : BOOL := FALSE;
  SH_DX_Done               : BOOL := FALSE;
  SH_DR_MoveRequested      : BOOL := FALSE;
  SH_DR_Done               : BOOL := FALSE;
  SH_Z_MoveRequested       : BOOL := FALSE;
  SH_Z_Done                : BOOL := FALSE;

  (* Rotation Arm placeholders - sonraki aşamalar için *)
  RA_W_MoveRequested       : BOOL := FALSE;
  RA_W_Done                : BOOL := FALSE;
  RotArm_ReadyForLoad      : BOOL := FALSE;
  RotArm_RotationDone      : BOOL := FALSE;
  RotArm_PunchReleased     : BOOL := FALSE;
  RotRequired              : BOOL := FALSE;
END_VAR

VAR
  risingStart              : BOOL;
  i                        : INT;
  foundFirst               : BOOL;
  punchDoneRising          : BOOL;
  lastPunchCompleted       : BOOL := FALSE;
  tmpInt                   : INT;

  AutoRunEnable            : BOOL;

  DelayFor_ClampOff        : BOOL;
  DelayFor_ClampOnn        : BOOL;
  ClampOff_Timer           : TON;
  ClampOn_Timer            : TON;
  tmpInt2:INT;
  k:INT;
  l:INT;
END_VAR
(* ------------------------------------------------------------ *)
(* Combined enable: must be on Auto page AND General_Start TRUE *)
(* ------------------------------------------------------------ *)
AutoRunEnable := (General_Start AND Automatic_Page);
ClampOff_Timer(PT:=TO_TIME(TO_INT(Retain_GeneralParameters[4]*1000)));
ClampOn_Timer(PT:=TO_TIME(TO_INT(Retain_GeneralParameters[3]*1000)));
(* ------------------------------------------------------------ *)
(* Edge detection                                               *)
(* ------------------------------------------------------------ *)
risingStart := (LoadStartCmd AND NOT lastStartCmd);


(* ------------------------------------------------------------ *)
(* HARD ABORT: reset all FB-internal state (lists are untouched) *)
(* ------------------------------------------------------------ *)
IF Abort_Cycle THEN
  LoadStartAck := FALSE;
  LoadBusy := FALSE;
  LoadError := FALSE;
  LoadDone := FALSE;

  lastStartCmd := FALSE;
  initiated := FALSE;

  Current_PunchIndex := 0;
  Current_PunchLoaded := FALSE;
  LastInitializedPunch := 0;

  LoadedPunchCount := 0;

  Local_Shuttle_State := 0;
  Local_WareHouse_State := 0;
  Local_RotArm_State := 0;

  PLoad_WH_SubStates := PLoad_WH_SubState.PLD_IDLE;
  PLoad_SH_SubStates := PLoad_SH_SubState.SH_IDLE;
  PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_IDLE;

  WH_TargetDrawerIdx := 0;
  WH_TargetRackID := 0;
  WH_NeedRackBring := FALSE;
  WH_NeedRackChange := FALSE;
  WH_SameRack := FALSE;

  WH_R_MoveRequested := FALSE;
  WH_X_MoveRequested := FALSE;
  WH_R_Done := FALSE;
  WH_X_Done := FALSE;
  WH_WaitRackInPos := FALSE;

  WH_R_LoadRequested := FALSE;
  WH_R_LoadDone := FALSE;
  R_Not_Safety_Zone := FALSE;
  Rack_Presence_Error := FALSE;

  SH_DX_MoveRequested := FALSE;
  SH_DX_Done := FALSE;
  SH_DR_MoveRequested := FALSE;
  SH_DR_Done := FALSE;
  SH_Z_MoveRequested := FALSE;
  SH_Z_Done := FALSE;

  RotArm_ReadyForLoad := FALSE;
  RotArm_RotationDone := FALSE;
  RotArm_PunchReleased := FALSE;
  RA_W_MoveRequested := FALSE;
  RA_W_Done := FALSE;
  RotRequired := FALSE;

  arAxes[EN_Axes.ATC_DR].AxisAutoSetVelocity := Retain_DRParameters[1];

  arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_DX].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_DR].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_Z].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := FALSE;

  RETURN;
END_IF


(* ------------------------------------------------------------ *)
(* PAUSE: freeze states, cut requests                           *)
(* ------------------------------------------------------------ *)
IF NOT AutoRunEnable AND NOT PLC_PRG.HMI.Drawer_Demand_Enable THEN
  arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_DX].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_DR].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_Z].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := FALSE;

  lastStartCmd := LoadStartCmd;
  RETURN;
END_IF


(* ------------------------------------------------------------ *)
(* SOFT RESET: do not reset states/substates; only cut requests *)
(* ------------------------------------------------------------ *)
IF Reset THEN
  arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_DX].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_DR].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_Z].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := FALSE;
END_IF


(* ------------------------------------------------------------ *)
(* Start handshake + init punch count + set initial punch index *)
(* ------------------------------------------------------------ *)
IF risingStart THEN
  initiated := TRUE;
  LoadBusy := TRUE;
  LoadError := FALSE;
  LoadDone := FALSE;

  LoadedPunchCount := 0;
  FOR i := 1 TO 15 DO
    IF Retain_LoadPunchList[i,3] <> '' THEN
      LoadedPunchCount := i;
    ELSE
      EXIT;
    END_IF;
  END_FOR;

  IF LoadedPunchCount = 0 THEN
    Current_PunchIndex := 0;
    Current_PunchLoaded := FALSE;
    LoadDone := TRUE;
    initiated := FALSE;

    Local_WareHouse_State := 0;
    Local_Shuttle_State := 0;
    Local_RotArm_State := 0;

    PLoad_WH_SubStates := PLoad_WH_SubState.PLD_IDLE;
    PLoad_SH_SubStates := PLoad_SH_SubState.SH_IDLE;
    PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_IDLE;

  ELSE
    Current_PunchIndex := 1;
    Current_PunchLoaded := TRUE;
    LoadDone := FALSE;

    Local_WareHouse_State := WareHouse_States.L_PDrawer_pickup_start;
    PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_INIT;

    Local_Shuttle_State := Shuttle_States.L_Shuttle_starts_PPick_up;
    PLoad_SH_SubStates := PLoad_SH_SubState.PLD_L10_INIT;

    Local_RotArm_State := RotationArm_States.L_RotArm_Init_And_Home;
    PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INIT_CHECK_POSITION;
  END_IF;

  LoadStartAck := TRUE;

ELSIF initiated AND LoadStartCmd THEN
  LoadStartAck := TRUE;
  LoadBusy := TRUE;

ELSIF initiated AND NOT LoadStartCmd THEN
  LoadStartAck := FALSE;
  LoadBusy := TRUE;

ELSE
  LoadStartAck := FALSE;
  IF NOT initiated THEN
    LoadBusy := FALSE;
  END_IF;
END_IF


(* ------------------------------------------------------------ *)
(* New punch initialization                                     *)
(* ------------------------------------------------------------ *)
IF (Current_PunchIndex <> 0) AND (Current_PunchIndex <> LastInitializedPunch) THEN
  Local_WareHouse_State := WareHouse_States.L_PDrawer_pickup_start;
  Local_Shuttle_State := Shuttle_States.L_Shuttle_starts_PPick_up;
  LastInitializedPunch := Current_PunchIndex;

  WH_TargetDrawerIdx := 0;
  WH_TargetRackID := 0;
  WH_NeedRackBring := FALSE;
  WH_NeedRackChange := FALSE;
  WH_SameRack := FALSE;

  WH_R_MoveRequested := FALSE;
  WH_X_MoveRequested := FALSE;
  WH_R_Done := FALSE;
  WH_X_Done := FALSE;
  WH_WaitRackInPos := FALSE;

  R_Not_Safety_Zone := FALSE;
  WH_R_LoadRequested := FALSE;
  WH_R_LoadDone := FALSE;
  Rack_Presence_Error := FALSE;

  SH_DX_MoveRequested := FALSE;
  SH_DX_Done := FALSE;
  SH_DR_MoveRequested := FALSE;
  SH_DR_Done := FALSE;
  SH_Z_MoveRequested := FALSE;
  SH_Z_Done := FALSE;

  RotRequired := FALSE;
  RotArm_RotationDone := FALSE;
  RotArm_PunchReleased := FALSE;

  PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_INIT;
  PLoad_SH_SubStates := PLoad_SH_SubState.PLD_L10_INIT;
END_IF


(* ============================================================ *)
(* Rotation Arm Engine                                          *)
(* ============================================================ *)
IF LoadStartCmd THEN

  CASE Local_RotArm_State OF

    RotationArm_States.L_RotArm_Init_And_Home:

      CASE PLoad_RotArm_SubStates OF

        PLoad_RotArm_SubState.ROT_IDLE:
          PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INIT_CHECK_POSITION;

        PLoad_RotArm_SubState.ROT_L_INIT_CHECK_POSITION:
          RotArm_ReadyForLoad := FALSE;
          RotArm_RotationDone := FALSE;
          RotArm_PunchReleased := FALSE;
          RA_W_MoveRequested := FALSE;
          RA_W_Done := FALSE;

          IF Fmsc_PLC_IO.Rotation_Arm_Inside THEN
            IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - 90.0) <= WH_Eps THEN
              PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INIT_MOVE_OUTSIDE_CMD;
            ELSE
              PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INIT_W_TO_90_REQ;
            END_IF;

          ELSIF Fmsc_PLC_IO.Rotation_Arm_Outside THEN
            IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - 180.0) <= WH_Eps THEN
              PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INIT_LOCK_CMD;
            ELSE
              PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INIT_W_TO_180_REQ;
            END_IF;

          ELSE
            LoadError := TRUE;
          END_IF;

        PLoad_RotArm_SubState.ROT_L_INIT_W_TO_90_REQ:
          arAxes[EN_Axes.ATC_W].TargetPos := 90.0;
          arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := TRUE;
          RA_W_MoveRequested := TRUE;
          RA_W_Done := FALSE;
          PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INIT_W_TO_90_WAIT;

        PLoad_RotArm_SubState.ROT_L_INIT_W_TO_90_WAIT:
          IF (NOT arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs) AND
             (ABS(arAxes[EN_Axes.ATC_W].ActualPos - arAxes[EN_Axes.ATC_W].TargetPos) > WH_Eps) THEN
            PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INIT_W_TO_90_REQ;
          END_IF;

          IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - arAxes[EN_Axes.ATC_W].TargetPos) <= WH_Eps THEN
            arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := FALSE;
            RA_W_MoveRequested := FALSE;
            RA_W_Done := TRUE;
            PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INIT_MOVE_OUTSIDE_CMD;
          END_IF;

        PLoad_RotArm_SubState.ROT_L_INIT_MOVE_OUTSIDE_CMD:
          Fmsc_PLC_IO.Rotation_Arm_Move_Inside_Valve := FALSE;
          Fmsc_PLC_IO.Rotation_Arm_Move_Outside_Valve := TRUE;
          PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INIT_WAIT_OUTSIDE;

        PLoad_RotArm_SubState.ROT_L_INIT_WAIT_OUTSIDE:
          IF Fmsc_PLC_IO.Rotation_Arm_Outside THEN
            Fmsc_PLC_IO.Rotation_Arm_Move_Outside_Valve := FALSE;

            IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - 180.0) <= WH_Eps THEN
              PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INIT_LOCK_CMD;
            ELSE
              PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INIT_W_TO_180_REQ;
            END_IF;
          END_IF;

        PLoad_RotArm_SubState.ROT_L_INIT_W_TO_180_REQ:
          arAxes[EN_Axes.ATC_W].TargetPos := 180.0;
          arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := TRUE;
          RA_W_MoveRequested := TRUE;
          RA_W_Done := FALSE;
          PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INIT_W_TO_180_WAIT;

        PLoad_RotArm_SubState.ROT_L_INIT_W_TO_180_WAIT:
          IF (NOT arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs) AND
             (ABS(arAxes[EN_Axes.ATC_W].ActualPos - arAxes[EN_Axes.ATC_W].TargetPos) > WH_Eps) THEN
            PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INIT_W_TO_180_REQ;
          END_IF;

          IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - arAxes[EN_Axes.ATC_W].TargetPos) <= WH_Eps THEN
            arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := FALSE;
            RA_W_MoveRequested := FALSE;
            RA_W_Done := TRUE;
            PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INIT_LOCK_CMD;
          END_IF;

        PLoad_RotArm_SubState.ROT_L_INIT_LOCK_CMD:
          Fmsc_PLC_IO.Rotation_Arm_Unlocking_Valve := FALSE;
          Fmsc_PLC_IO.Rotation_Arm_Locking_Valve := TRUE;
          PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INIT_WAIT_LOCKED;

        PLoad_RotArm_SubState.ROT_L_INIT_WAIT_LOCKED:
          IF Fmsc_PLC_IO.Rotation_Arm_Locked_CH2 AND Fmsc_PLC_IO.Rotation_Arm_Locked_CH3 THEN
            RotArm_ReadyForLoad := TRUE;
            Local_RotArm_State := RotationArm_States.L_RotArm_Ready_For_Load;
            PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_READY;
          END_IF;

      ELSE
        PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INIT_CHECK_POSITION;
      END_CASE;


    RotationArm_States.L_RotArm_Ready_For_Load:

      CASE PLoad_RotArm_SubStates OF

        PLoad_RotArm_SubState.ROT_L_READY:
          RotArm_ReadyForLoad := TRUE;

          IF RotRequired THEN
            Local_RotArm_State := RotationArm_States.L_RotArm_Rotation_Interrupt_Active;
            PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_WAIT_PUNCH_ON_ARM;
          END_IF;

      ELSE
        PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_READY;
      END_CASE;


    RotationArm_States.L_RotArm_Rotation_Interrupt_Active:

      CASE PLoad_RotArm_SubStates OF

        PLoad_RotArm_SubState.ROT_L_INT_WAIT_PUNCH_ON_ARM:
          RotArm_ReadyForLoad := FALSE;
          RotArm_RotationDone := FALSE;
          RotArm_PunchReleased := FALSE;

          IF Fmsc_PLC_IO.Punch_Locked_on_Rotation_Arm THEN
            PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_WAIT_SHUTTLE_CLEAR;
          END_IF;

        PLoad_RotArm_SubState.ROT_L_INT_WAIT_SHUTTLE_CLEAR:
          IF (PLoad_SH_SubStates = PLoad_SH_SubState.PLD_L13_DONE) THEN
            PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_ARM_UNLOCK_CMD;
          END_IF;

        PLoad_RotArm_SubState.ROT_L_INT_ARM_UNLOCK_CMD:
          Fmsc_PLC_IO.Rotation_Arm_Unlocking_Valve := TRUE;
          Fmsc_PLC_IO.Rotation_Arm_Locking_Valve := FALSE;
          PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_WAIT_ARM_UNLOCKED;

        PLoad_RotArm_SubState.ROT_L_INT_WAIT_ARM_UNLOCKED:
          IF Fmsc_PLC_IO.Rotation_Arm_Unlocked_CH2 AND Fmsc_PLC_IO.Rotation_Arm_Unlocked_CH3 THEN
            PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_W_TO_ROTATED_REQ;
          END_IF;

        PLoad_RotArm_SubState.ROT_L_INT_W_TO_ROTATED_REQ:
          arAxes[EN_Axes.ATC_W].TargetPos := 0.0;
          arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := TRUE;
          RA_W_MoveRequested := TRUE;
          RA_W_Done := FALSE;
          PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_W_TO_ROTATED_WAIT;

        PLoad_RotArm_SubState.ROT_L_INT_W_TO_ROTATED_WAIT:
          IF (NOT arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs) AND
             (ABS(arAxes[EN_Axes.ATC_W].ActualPos - arAxes[EN_Axes.ATC_W].TargetPos) > WH_Eps) THEN
            PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_W_TO_ROTATED_REQ;
          END_IF;

          IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - arAxes[EN_Axes.ATC_W].TargetPos) <= WH_Eps THEN
            arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := FALSE;
            RA_W_MoveRequested := FALSE;
            RA_W_Done := TRUE;
            PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_ARM_LOCK_CMD;
          END_IF;

        PLoad_RotArm_SubState.ROT_L_INT_ARM_LOCK_CMD:
          Fmsc_PLC_IO.Rotation_Arm_Locking_Valve := TRUE;
          Fmsc_PLC_IO.Rotation_Arm_Unlocking_Valve := FALSE;
          PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_WAIT_ARM_LOCKED;

        PLoad_RotArm_SubState.ROT_L_INT_WAIT_ARM_LOCKED:
          IF Fmsc_PLC_IO.Rotation_Arm_Locked_CH2 AND Fmsc_PLC_IO.Rotation_Arm_Locked_CH3 THEN
            RotArm_RotationDone := TRUE;
            PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_WAIT_SHUTTLE_PICKBACK;
          END_IF;

        PLoad_RotArm_SubState.ROT_L_INT_WAIT_SHUTTLE_PICKBACK:
          IF (PLoad_SH_SubStates = PLoad_SH_SubState.PLD_L15_WAIT_ROTARM_RELEASED) THEN
            PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_UNLOCK_CMD;
          END_IF;

        PLoad_RotArm_SubState.ROT_L_INT_UNLOCK_CMD:
          Fmsc_PLC_IO.Rotation_Arm_Punch_Unlock_Valve := TRUE;
          Fmsc_PLC_IO.Rotation_Arm_Punch_Lock_Valve := FALSE;
          PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_WAIT_PUNCH_RELEASED;

        PLoad_RotArm_SubState.ROT_L_INT_WAIT_PUNCH_RELEASED:
          IF NOT Fmsc_PLC_IO.Punch_Locked_on_Rotation_Arm THEN
            RotArm_PunchReleased := TRUE;
            PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_RETURN_UNLOCK_CMD;
          END_IF;

        PLoad_RotArm_SubState.ROT_L_INT_RETURN_UNLOCK_CMD:
          Fmsc_PLC_IO.Rotation_Arm_Unlocking_Valve := TRUE;
          Fmsc_PLC_IO.Rotation_Arm_Locking_Valve := FALSE;
          PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_RETURN_WAIT_Z_SAFE_FOR_READY;

        PLoad_RotArm_SubState.ROT_L_INT_RETURN_WAIT_Z_SAFE_FOR_READY:
          IF arAxes[EN_Axes.ATC_Z].ActualPos > TO_LREAL(Retain_ZParameters[9]) THEN
            PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_RETURN_WAIT_UNLOCKED;
          END_IF;

        PLoad_RotArm_SubState.ROT_L_INT_RETURN_WAIT_UNLOCKED:
          IF Fmsc_PLC_IO.Rotation_Arm_Unlocked_CH2 AND Fmsc_PLC_IO.Rotation_Arm_Unlocked_CH3 THEN
            PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_RETURN_W_TO_180_REQ;
          END_IF;

        PLoad_RotArm_SubState.ROT_L_INT_RETURN_W_TO_180_REQ:
          arAxes[EN_Axes.ATC_W].TargetPos := 180.0;
          arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := TRUE;
          RA_W_MoveRequested := TRUE;
          RA_W_Done := FALSE;
          PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_RETURN_W_TO_180_WAIT;

        PLoad_RotArm_SubState.ROT_L_INT_RETURN_W_TO_180_WAIT:
          IF (NOT arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs) AND
             (ABS(arAxes[EN_Axes.ATC_W].ActualPos - arAxes[EN_Axes.ATC_W].TargetPos) > WH_Eps) THEN
            PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_RETURN_W_TO_180_REQ;
          END_IF;

          IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - arAxes[EN_Axes.ATC_W].TargetPos) <= WH_Eps THEN
            arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := FALSE;
            RA_W_MoveRequested := FALSE;
            RA_W_Done := TRUE;
            PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_RETURN_LOCK_CMD;
          END_IF;

        PLoad_RotArm_SubState.ROT_L_INT_RETURN_LOCK_CMD:
          Fmsc_PLC_IO.Rotation_Arm_Locking_Valve := TRUE;
          Fmsc_PLC_IO.Rotation_Arm_Unlocking_Valve := FALSE;
          PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_RETURN_WAIT_LOCKED;

        PLoad_RotArm_SubState.ROT_L_INT_RETURN_WAIT_LOCKED:
          IF Fmsc_PLC_IO.Rotation_Arm_Locked_CH2 AND Fmsc_PLC_IO.Rotation_Arm_Locked_CH3 THEN
            PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_DONE;
          END_IF;

        PLoad_RotArm_SubState.ROT_L_INT_DONE:
          RotRequired := FALSE;
          RotArm_ReadyForLoad := TRUE;
          Local_RotArm_State := RotationArm_States.L_RotArm_Ready_For_Load;
          PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_READY;

      ELSE
        PLoad_RotArm_SubStates := PLoad_RotArm_SubState.ROT_L_INT_WAIT_PUNCH_ON_ARM;
      END_CASE;

  ELSE
    ;
  END_CASE;

END_IF


(* ============================================================ *)
(* Warehouse macro states 9..12: LOAD punches (WH engine)       *)
(* ============================================================ *)
IF (Local_WareHouse_State = WareHouse_States.L_PDrawer_pickup_start) OR
   (Local_WareHouse_State = WareHouse_States.L_Punches_rack_taken) OR
   (Local_WareHouse_State = WareHouse_States.L_PDrawer_pickup_Pos_reached) OR
   (Local_WareHouse_State = WareHouse_States.L_PPickup_from_drawer_allowed) THEN

  IF Current_PunchIndex = 0 THEN
    LoadError := TRUE;

  ELSE

    CASE PLoad_WH_SubStates OF

      PLoad_WH_SubState.PLD_IDLE:
        PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_INIT;

      PLoad_WH_SubState.PLD_L09_INIT:
        WH_TargetDrawerIdx := 0;
        WH_TargetRackID := 0;
        WH_NeedRackBring := FALSE;
        WH_NeedRackChange := FALSE;
        WH_SameRack := FALSE;

        PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_READ_TARGETS;

      PLoad_WH_SubState.PLD_L09_READ_TARGETS:
        IF Retain_ATCLoadPunchList[Current_PunchIndex,2] <> '' THEN
          WH_TargetDrawerIdx := STRING_TO_INT(Retain_ATCLoadPunchList[Current_PunchIndex,2]);
        ELSE
          WH_TargetDrawerIdx := 0;
        END_IF;

        IF WH_TargetDrawerIdx >= 1 THEN
          tmpInt := WH_TargetDrawerIdx - 1;
          WH_TargetRackID := TO_INT(tmpInt / 4) + 1;
        ELSE
          WH_TargetRackID := 0;
        END_IF;

        IF Current_Rack_Type = 0 THEN
          WH_NeedRackBring := TRUE;
          WH_NeedRackChange := FALSE;
          WH_SameRack := FALSE;

        ELSIF Current_Rack_Type = 1 THEN
          WH_NeedRackBring := FALSE;
          IF TO_INT(Current_Rack_ID) = WH_TargetRackID THEN
            WH_SameRack := TRUE;
            WH_NeedRackChange := FALSE;
          ELSE
            WH_SameRack := FALSE;
            WH_NeedRackChange := TRUE;
          END_IF

        ELSIF Current_Rack_Type = 2 THEN
          WH_NeedRackBring := FALSE;
          WH_NeedRackChange := TRUE;
          WH_SameRack := FALSE;
        END_IF

        IF WH_NeedRackBring THEN
          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_UNLOCK_CHECK;

        ELSIF WH_SameRack THEN
          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_SAMERACK_DRAWER_CHECK;

        ELSIF WH_NeedRackChange THEN
          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_RACKCHANGE_UNLOCK_CHECK;

        ELSE
          LoadError := TRUE;
        END_IF;

      PLoad_WH_SubState.PLD_L09_SAMERACK_DRAWER_CHECK:
        IF Current_Drawer_ID = WH_TargetDrawerIdx THEN
          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L12_DONE;
        ELSE
          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_SAMERACK_R_TO_DRAWER_REQ;
        END_IF;

      PLoad_WH_SubState.PLD_L09_SAMERACK_R_TO_DRAWER_REQ:
        IF (WH_TargetDrawerIdx >= 1) AND (WH_TargetDrawerIdx <= 16) THEN

          IF ABS(arAxes[EN_Axes.ATC_R].ActualPos - TO_LREAL(pDrawerParam1[WH_TargetDrawerIdx])) <= WH_Eps THEN
            arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
            WH_R_MoveRequested := FALSE;
            WH_R_Done := TRUE;
            Current_Drawer_ID := WH_TargetDrawerIdx;

            PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L12_DONE;
          ELSE
            arAxes[EN_Axes.ATC_R].TargetPos := TO_LREAL(pDrawerParam1[WH_TargetDrawerIdx]);
            arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := TRUE;
            WH_R_MoveRequested := TRUE;
            WH_R_Done := FALSE;

            PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_SAMERACK_R_TO_DRAWER_WAIT;
          END_IF;

        ELSE
          LoadError := TRUE;
        END_IF;

      PLoad_WH_SubState.PLD_L09_SAMERACK_R_TO_DRAWER_WAIT:
        IF (NOT arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs) AND
           (ABS(arAxes[EN_Axes.ATC_R].ActualPos - arAxes[EN_Axes.ATC_R].TargetPos) > WH_Eps) THEN
          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_SAMERACK_R_TO_DRAWER_REQ;
        END_IF;

        IF ABS(arAxes[EN_Axes.ATC_R].ActualPos - arAxes[EN_Axes.ATC_R].TargetPos) <= WH_Eps THEN
          arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
          WH_R_MoveRequested := FALSE;
          WH_R_Done := TRUE;
          Current_Drawer_ID := WH_TargetDrawerIdx;

          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L12_DONE;
        END_IF;

      PLoad_WH_SubState.PLD_L09_RACKCHANGE_UNLOCK_CHECK:
        IF NOT (Fmsc_PLC_IO.Carriage_Unlocked_Left AND Fmsc_PLC_IO.Carriage_Unlocked_Right) THEN
          Fmsc_PLC_IO.Carriage_UnLocking_Valve := TRUE;
          FMSC_PLC_IO.Carriage_Locking_Valve := FALSE;
        ELSE
          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_RACKCHANGE_R_TO_SAFE_REQ;
        END_IF;

      PLoad_WH_SubState.PLD_L09_RACKCHANGE_R_TO_SAFE_REQ:
        arAxes[EN_Axes.ATC_R].TargetPos := TO_LREAL(Retain_RParameters[3]);
        arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := TRUE;
        WH_R_MoveRequested := TRUE;
        WH_R_Done := FALSE;

        PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_RACKCHANGE_R_TO_SAFE_WAIT;

      PLoad_WH_SubState.PLD_L09_RACKCHANGE_R_TO_SAFE_WAIT:
        IF (NOT arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs) AND
           (ABS(arAxes[EN_Axes.ATC_R].ActualPos - arAxes[EN_Axes.ATC_R].TargetPos) > WH_Eps) THEN
          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_RACKCHANGE_R_TO_SAFE_REQ;
        END_IF;

        IF ABS(arAxes[EN_Axes.ATC_R].ActualPos - arAxes[EN_Axes.ATC_R].TargetPos) <= WH_Eps THEN
          arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
          WH_R_MoveRequested := FALSE;
          WH_R_Done := TRUE;

          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_RACKCHANGE_X_TO_CURRENT_DRAWER_REQ;
        END_IF;

      PLoad_WH_SubState.PLD_L09_RACKCHANGE_X_TO_CURRENT_DRAWER_REQ:
        IF (Current_Drawer_ID >= 1) AND (Current_Drawer_ID <= 16) THEN

          IF Current_Rack_Type = 2 THEN
            IF ABS(arAxes[EN_Axes.ATC_X].ActualPos - TO_LREAL(DDrawerParam2[TO_INT(Current_Drawer_ID)])) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := FALSE;
              WH_X_MoveRequested := FALSE;
              WH_X_Done := TRUE;

              PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_RACKCHANGE_R_TO_DROP_REQ;
            ELSE
              arAxes[EN_Axes.ATC_X].TargetPos := TO_LREAL(DDrawerParam2[TO_INT(Current_Drawer_ID)]);
              arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := TRUE;
              WH_X_MoveRequested := TRUE;
              WH_X_Done := FALSE;

              PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_RACKCHANGE_X_TO_CURRENT_DRAWER_WAIT;
            END_IF;
          ELSE
            IF ABS(arAxes[EN_Axes.ATC_X].ActualPos - TO_LREAL(pDrawerParam2[TO_INT(Current_Drawer_ID)])) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := FALSE;
              WH_X_MoveRequested := FALSE;
              WH_X_Done := TRUE;

              PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_RACKCHANGE_R_TO_DROP_REQ;
            ELSE
              arAxes[EN_Axes.ATC_X].TargetPos := TO_LREAL(pDrawerParam2[TO_INT(Current_Drawer_ID)]);
              arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := TRUE;
              WH_X_MoveRequested := TRUE;
              WH_X_Done := FALSE;

              PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_RACKCHANGE_X_TO_CURRENT_DRAWER_WAIT;
            END_IF;
          END_IF;

        ELSE
          LoadError := TRUE;
        END_IF;

      PLoad_WH_SubState.PLD_L09_RACKCHANGE_X_TO_CURRENT_DRAWER_WAIT:
        IF (NOT arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs) AND
           (ABS(arAxes[EN_Axes.ATC_X].ActualPos - arAxes[EN_Axes.ATC_X].TargetPos) > WH_Eps) THEN
          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_RACKCHANGE_X_TO_CURRENT_DRAWER_REQ;
        END_IF;

        IF ABS(arAxes[EN_Axes.ATC_X].ActualPos - arAxes[EN_Axes.ATC_X].TargetPos) <= WH_Eps THEN
          arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := FALSE;
          WH_X_MoveRequested := FALSE;
          WH_X_Done := TRUE;

          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_RACKCHANGE_R_TO_DROP_REQ;
        END_IF;

      PLoad_WH_SubState.PLD_L09_RACKCHANGE_R_TO_DROP_REQ:
        arAxes[EN_Axes.ATC_R].TargetPos := -10.0;
        arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := TRUE;
        WH_R_MoveRequested := TRUE;
        WH_R_Done := FALSE;

        PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_RACKCHANGE_R_TO_DROP_WAIT;

      PLoad_WH_SubState.PLD_L09_RACKCHANGE_R_TO_DROP_WAIT:
        IF (NOT arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs) AND
           (ABS(arAxes[EN_Axes.ATC_R].ActualPos - arAxes[EN_Axes.ATC_R].TargetPos) > WH_Eps) THEN
          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_RACKCHANGE_R_TO_DROP_REQ;
        END_IF;

        IF ABS(arAxes[EN_Axes.ATC_R].ActualPos - arAxes[EN_Axes.ATC_R].TargetPos) <= WH_Eps THEN
          arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
          WH_R_MoveRequested := FALSE;
          WH_R_Done := TRUE;

          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_RACKCHANGE_WAIT_RACK_RELEASED;
        END_IF;

      PLoad_WH_SubState.PLD_L09_RACKCHANGE_WAIT_RACK_RELEASED:
        IF NOT Fmsc_PLC_IO.Rack_Presence THEN
          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_RACKCHANGE_CLEAR_CURRENT_INFO;
        END_IF;

      PLoad_WH_SubState.PLD_L09_RACKCHANGE_CLEAR_CURRENT_INFO:
        Current_Rack_ID := 0;
        Current_Rack_Type := 0;
        Current_Drawer_ID := 0;

        WH_NeedRackChange := FALSE;
        WH_NeedRackBring := TRUE;
        WH_SameRack := FALSE;

        WH_R_MoveRequested := FALSE;
        WH_X_MoveRequested := FALSE;
        WH_R_Done := FALSE;
        WH_X_Done := FALSE;
        WH_WaitRackInPos := FALSE;

        WH_R_LoadRequested := FALSE;
        WH_R_LoadDone := FALSE;
        R_Not_Safety_Zone := FALSE;
        Rack_Presence_Error := FALSE;

        PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_UNLOCK_CHECK;

      PLoad_WH_SubState.PLD_L09_UNLOCK_CHECK:
        IF NOT (Fmsc_PLC_IO.Carriage_Unlocked_Left AND Fmsc_PLC_IO.Carriage_Unlocked_Right) THEN
          Fmsc_PLC_IO.Carriage_UnLocking_Valve := TRUE;
        ELSE
          Fmsc_PLC_IO.Carriage_UnLocking_Valve := FALSE;

          IF R_Not_Safety_Zone THEN
            PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_WAIT_R_SAFEZONE;
          ELSE
            PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_R_TO_PICKUP_REQ;
          END_IF
        END_IF;

      PLoad_WH_SubState.PLD_L09_WAIT_R_SAFEZONE:
        IF arAxes[EN_Axes.ATC_R].ActualPos < TO_LREAL(Retain_RParameters[7]) THEN
          R_Not_Safety_Zone := FALSE;
          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_R_TO_PICKUP_REQ;
        END_IF;

      PLoad_WH_SubState.PLD_L09_R_TO_PICKUP_REQ:
        IF arAxes[EN_Axes.ATC_R].ActualPos < TO_LREAL(Retain_RParameters[7]) THEN
          arAxes[EN_Axes.ATC_R].TargetPos := -10.0;
          arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := TRUE;
          WH_R_MoveRequested := TRUE;
          WH_R_Done := FALSE;

          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_R_TO_PICKUP_WAIT;
        ELSE
          R_Not_Safety_Zone := TRUE;
          WH_R_MoveRequested := FALSE;
          WH_R_Done := FALSE;
          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_WAIT_R_SAFEZONE;
        END_IF;

      PLoad_WH_SubState.PLD_L09_R_TO_PICKUP_WAIT:
        IF (NOT arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs) AND
           (ABS(arAxes[EN_Axes.ATC_R].ActualPos - (-10.0)) > WH_Eps) THEN
          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_R_TO_PICKUP_REQ;
        END_IF;

        IF ABS(arAxes[EN_Axes.ATC_R].ActualPos - (-10.0)) <= WH_Eps THEN
          arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
          WH_R_MoveRequested := FALSE;
          WH_R_Done := TRUE;

          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_X_TO_DRAWER_REQ;
        END_IF;

      PLoad_WH_SubState.PLD_L09_X_TO_DRAWER_REQ:
        IF WH_TargetDrawerIdx >= 1 THEN

          IF ABS(arAxes[EN_Axes.ATC_X].ActualPos - TO_LREAL(pDrawerParam2[WH_TargetDrawerIdx])) <= WH_Eps THEN
            WH_X_Done := TRUE;
            WH_X_MoveRequested := FALSE;
            PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_R_LIFT_REQ;
          ELSE
            arAxes[EN_Axes.ATC_X].TargetPos := TO_LREAL(pDrawerParam2[WH_TargetDrawerIdx]);
            arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := TRUE;
            WH_X_MoveRequested := TRUE;
            WH_X_Done := FALSE;

            PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_X_TO_DRAWER_WAIT;
          END_IF

        ELSE
          LoadError := TRUE;
        END_IF;

      PLoad_WH_SubState.PLD_L09_X_TO_DRAWER_WAIT:
        IF (NOT arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs) AND
           (ABS(arAxes[EN_Axes.ATC_X].ActualPos - arAxes[EN_Axes.ATC_X].TargetPos) > WH_Eps) THEN
          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_X_TO_DRAWER_REQ;
        END_IF;

        IF ABS(arAxes[EN_Axes.ATC_X].ActualPos - arAxes[EN_Axes.ATC_X].TargetPos) <= WH_Eps THEN
          arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := FALSE;
          WH_X_MoveRequested := FALSE;
          WH_X_Done := TRUE;

          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_R_LIFT_REQ;
        END_IF;

      PLoad_WH_SubState.PLD_L09_R_LIFT_REQ:
        arAxes[EN_Axes.ATC_R].TargetPos := TO_LREAL(Retain_RParameters[3]);
        arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := TRUE;
        WH_R_LoadRequested := TRUE;
        WH_R_LoadDone := FALSE;
        Rack_Presence_Error := FALSE;

        PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_R_LIFT_WAIT;

      PLoad_WH_SubState.PLD_L09_R_LIFT_WAIT:
        IF (NOT arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs) AND
           (ABS(arAxes[EN_Axes.ATC_R].ActualPos - arAxes[EN_Axes.ATC_R].TargetPos) > WH_Eps) THEN
          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_R_LIFT_REQ;
        END_IF;

        IF arAxes[EN_Axes.ATC_R].ActualPos >= TO_LREAL(Retain_RParameters[7]) THEN

          IF Fmsc_PLC_IO.Rack_Presence THEN
            Current_Rack_ID := WH_TargetRackID;
            Current_Rack_Type := 1;
            Current_Drawer_ID := WH_TargetDrawerIdx;
          ELSE
            arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
            WH_R_LoadRequested := FALSE;
            WH_R_LoadDone := FALSE;
            Rack_Presence_Error := TRUE;

            General_Stop := TRUE;
            General_Start := FALSE;

            PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_ERROR_RACK_PRESENCE;
          END_IF

        END_IF

        IF ABS(arAxes[EN_Axes.ATC_R].ActualPos - arAxes[EN_Axes.ATC_R].TargetPos) <= WH_Eps THEN
          arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
          WH_R_LoadRequested := FALSE;
          WH_R_LoadDone := TRUE;
          Rack_Presence_Error := FALSE;

          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_WAIT_RACK_IN_POS;
        END_IF;

      PLoad_WH_SubState.PLD_L09_WAIT_RACK_IN_POS:
        WH_WaitRackInPos := (WH_R_LoadDone AND WH_X_Done);

        IF WH_WaitRackInPos THEN
          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_DONE;
        END_IF;

      PLoad_WH_SubState.PLD_L09_ERROR_RACK_PRESENCE:
        WH_WaitRackInPos := FALSE;

        IF arAxes[EN_Axes.ATC_R].ActualPos < TO_LREAL(Retain_RParameters[7]) THEN
          Rack_Presence_Error := FALSE;
          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L09_R_LIFT_REQ;
        END_IF;

      PLoad_WH_SubState.PLD_L09_DONE:
        Local_WareHouse_State := WareHouse_States.L_Punches_rack_taken;
        PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L10_INIT;

      PLoad_WH_SubState.PLD_L10_INIT:
        WH_X_MoveRequested := FALSE;
        WH_X_Done := FALSE;
        WH_R_MoveRequested := FALSE;
        WH_R_Done := FALSE;

        PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L10_X_TO_PICKPOS_REQ;

      PLoad_WH_SubState.PLD_L10_X_TO_PICKPOS_REQ:
        IF ABS(arAxes[EN_Axes.ATC_X].ActualPos - TO_LREAL(Retain_XParameters[2])) <= WH_Eps THEN
          arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := FALSE;
          WH_X_MoveRequested := FALSE;
          WH_X_Done := TRUE;
          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L10_R_TO_PICKPOS_REQ;
        ELSE
          arAxes[EN_Axes.ATC_X].TargetPos := TO_LREAL(Retain_XParameters[2]);
          arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := TRUE;
          WH_X_MoveRequested := TRUE;
          WH_X_Done := FALSE;

          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L10_X_TO_PICKPOS_WAIT;
        END_IF;

      PLoad_WH_SubState.PLD_L10_X_TO_PICKPOS_WAIT:
        IF (NOT arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs) AND
           (ABS(arAxes[EN_Axes.ATC_X].ActualPos - arAxes[EN_Axes.ATC_X].TargetPos) > WH_Eps) THEN
          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L10_X_TO_PICKPOS_REQ;
        END_IF;

        IF ABS(arAxes[EN_Axes.ATC_X].ActualPos - arAxes[EN_Axes.ATC_X].TargetPos) <= WH_Eps THEN
          arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := FALSE;
          WH_X_MoveRequested := FALSE;
          WH_X_Done := TRUE;

          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L10_R_TO_PICKPOS_REQ;
        END_IF;

      PLoad_WH_SubState.PLD_L10_R_TO_PICKPOS_REQ:
        IF (WH_TargetDrawerIdx >= 1) AND (WH_TargetDrawerIdx <= 16) THEN

          IF ABS(arAxes[EN_Axes.ATC_R].ActualPos - TO_LREAL(pDrawerParam1[WH_TargetDrawerIdx])) <= WH_Eps THEN
            arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
            WH_R_MoveRequested := FALSE;
            WH_R_Done := TRUE;
            PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L10_DONE;
          ELSE
            arAxes[EN_Axes.ATC_R].TargetPos := TO_LREAL(pDrawerParam1[WH_TargetDrawerIdx]);
            arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := TRUE;
            WH_R_MoveRequested := TRUE;
            WH_R_Done := FALSE;

            PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L10_R_TO_PICKPOS_WAIT;
          END_IF

        ELSE
          LoadError := TRUE;
        END_IF;

      PLoad_WH_SubState.PLD_L10_R_TO_PICKPOS_WAIT:
        IF (NOT arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs) AND
           (ABS(arAxes[EN_Axes.ATC_R].ActualPos - arAxes[EN_Axes.ATC_R].TargetPos) > WH_Eps) THEN
          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L10_R_TO_PICKPOS_REQ;
        END_IF;

        IF ABS(arAxes[EN_Axes.ATC_R].ActualPos - arAxes[EN_Axes.ATC_R].TargetPos) <= WH_Eps THEN
          arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
          WH_R_MoveRequested := FALSE;
          WH_R_Done := TRUE;

          PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L10_DONE;
        END_IF;

      PLoad_WH_SubState.PLD_L10_DONE:
        Local_WareHouse_State := WareHouse_States.L_PDrawer_pickup_Pos_reached;
        PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L11_INIT;

      PLoad_WH_SubState.PLD_L11_INIT:
        WH_X_MoveRequested := FALSE;
        WH_X_Done := FALSE;
        WH_R_MoveRequested := FALSE;
        WH_R_Done := FALSE;

        PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L11_LOCK_CMD;

      PLoad_WH_SubState.PLD_L11_LOCK_CMD:
        Fmsc_PLC_IO.Carriage_UnLocking_Valve := FALSE;
        Fmsc_PLC_IO.Carriage_Locking_Valve := TRUE;

        PLoad_WH_SubStates := PLoad_WH_SubState.PLD_L11_WAIT_LOCKED;

      PLoad_WH_SubState.PLD_L11_WAIT_LOCKED:
        IF (Fmsc_PLC_IO.Carriage_Locked_Left AND Fmsc_PLC_IO.Carriage_Locked_Right) THEN
