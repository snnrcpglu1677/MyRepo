FUNCTION_BLOCK PunchChanger

VAR_INPUT
	Trigger : BOOL ;
	Unload: BOOL;
	Abbort: BOOL;
END_VAR

VAR_OUTPUT RETAIN
  ValidityFlag   : BOOL;
  PerIndexStatus : ARRAY[1..15] OF PIndexStatus;
  PackageState   : PPackageState;
  NextActionHint : PNextActionHint;

  numSame  : INT;
  numDiff  : INT;
  newCount : INT;
  oldCount : INT;

  ErrorText : STRING(200);
  Error : BOOL;
  Busy  : BOOL;
  Done  : BOOL;
END_VAR

VAR RETAIN
  i : INT;
  j : INT;
  k : INT;
  assigned:BOOL;
  foundIndex : INT;

  usedSlot :ARRAY [1..16, 1..15] OF BOOL;
  usedSlotOld :ARRAY [1..16, 1..15] OF BOOL;   (* unload: ATC slot cozumleme icin *)
  OldAssignedDrawer:ARRAY[1..15] OF INT;
  OldAssignedSlot:ARRAY[1..15] OF INT;

  NewAssignedDrawer:ARRAY[1..15] OF INT;
  NewAssignedSlot   : ARRAY[1..15] OF INT;
  origNewIdx:INT;
  NewName : STRING(100);
  OldName : STRING(100);
  NewLen : STRING(100);
  NewLenReal : REAL;
  NewHorn : STRING(100);
  NewHornReal : REAL;
  OldLen : STRING(100);
  OldLenReal : REAL;
  OldHorn : STRING(100);
  NewRot : STRING(100);
  OldRot : STRING(100);

  eps : REAL := 0.001;

  foundLengthMatch : BOOL;
  hasLengthValue : BOOL;

  KeepList : ARRAY[1..15] OF BOOL;
  ReplaceList : ARRAY[1..15] OF BOOL;
  LoadOnlyList : ARRAY[1..15] OF BOOL;
  UnloadOnlyList : ARRAY[1..15] OF BOOL;
  OldMatched : ARRAY[1..15] OF BOOL;
  NewMatchedToOld : ARRAY[1..15] OF INT;

  UnloadTempCount : INT;
  UnloadTempOrigIndex : ARRAY[1..15] OF INT;
  UnloadTempDrawerIdx : ARRAY[1..15] OF INT;
  UnloadTempOrder : ARRAY[1..15] OF INT;

  LoadTempCount : INT;
  LoadTempOrigIndex : ARRAY[1..15] OF INT;
  LoadTempDrawerIdx : ARRAY[1..15] OF INT;
  LoadTempOrder : ARRAY[1..15] OF INT;

  KeepTempCount : INT;
  KeepTempOrigIndex : ARRAY[1..15] OF INT;
  KeepTempDrawerIdx : ARRAY[1..15] OF INT;
  KeepTempOrder : ARRAY[1..15] OF INT;

  a : INT;
  b : INT;
  minIdx : INT;
  tmpInt : INT;

  OldAvailCount : INT;
  OldAvailIndex : ARRAY[1..15] OF INT;

  c : INT;
  d : INT;
  e : INT;
  m : INT;
  UnloadCount: INT;
  LoadCount:INT;
  KeepCount :INT;
  lenStr : STRING(100);
  posStr : STRING(100);
  hornStr : STRING(100);
  lenRealLocal : REAL;
  hornRealLocal : REAL;

  neededCount : INT;
  availCount : INT;
  skipTriple : BOOL := FALSE;
  uniqCount : INT;
  uniqName : ARRAY[1..15] OF STRING(100);
  uniqLen  : ARRAY[1..15] OF STRING(100);
  uniqHorn : ARRAY[1..15] OF STRING(100);
  uniqNeeded : ARRAY[1..15] OF INT;

  (* Drag/Relocate helpers *)
  DragUnloadCount : INT;
  DragLoadCount : INT;
  RightmostDragUnloadPos : LREAL;
  RightmostDragLoadPos : LREAL;
  RelocateThreshold : LREAL;
  DragFlag : STRING(10);
  RelocateCount : INT;
  RelocateIdx : INT;

  (* Relocate threshold refinement *)
  MaxKeepAboveThreshold : LREAL;
  FoundKeepAboveThreshold : BOOL;

  (* Drawer / slot cozumleme helpers *)
  isDirectDrawer : BOOL;
  resolveFound : BOOL;
END_VAR

VAR
	eTmp : INT;
	dTmp : INT;
	tmpPos : LREAL;
	hasRelocateOverlap : BOOL;
	drawerTmp : INT;
	indicatorColumn : INT;
	indicatorPos : INT;
	indicatorValue : INT;
	indicatorLimit : INT;
	indicatorDigit : STRING(1);
	indicatorValid : BOOL;
	noActionMatchedOld : ARRAY[1..15] OF BOOL;
	noActionAllMatched : BOOL;
	noActionFound : BOOL;
END_VAR

(* ============================================================ *)
(* Drag / Direct kurali:                                         *)
(*   Drawer no 4/8/12/16  -> DIRECT ('0')                        *)
(*   Diger drawerlar      -> DRAG   ('1')                        *)
(*                                                               *)
(* Drawer no PFields / PFields_Old icinden OKUNMAZ (bos geliyor).*)
(*   - Load   : ATC eslestirmesinde secilen drawer (e)           *)
(*   - Unload : PunchUsed=TRUE olan ve isim/boy/horn eslesen slot*)
(* Drag drawerlarda load slotu BUYUKTEN kucuge taranir.          *)
(* ============================================================ *)


(* Abort: clear generated lists and transient changer state.
   PFields and PFields_Old are the physical/program source data and remain untouched. *)
IF Abbort THEN
  Busy := FALSE;
  Done := FALSE;
  Error := FALSE;
  ErrorText := '';
  ValidityFlag := FALSE;

  PackageState := PPackageState.INVALID;
  NextActionHint := PNextActionHint.INVALID_ABORT;

  numSame := 0;
  numDiff := 0;
  newCount := 0;
  oldCount := 0;
  UnloadCount := 0;
  LoadCount := 0;
  KeepCount := 0;
  UnloadTempCount := 0;
  LoadTempCount := 0;
  KeepTempCount := 0;
  OldAvailCount := 0;
  uniqCount := 0;
  availCount := 0;
  neededCount := 0;
  DragUnloadCount := 0;
  DragLoadCount := 0;
  RelocateCount := 0;
  RelocateIdx := 0;
  RightmostDragUnloadPos := 0.0;
  RightmostDragLoadPos := 0.0;
  RelocateThreshold := 0.0;
  RelocateTrashold_Value := 0.0;
  MaxKeepAboveThreshold := 0.0;
  FoundKeepAboveThreshold := FALSE;

  FOR i := 1 TO 15 DO
    PerIndexStatus[i] := PIndexStatus.SAME;
    KeepList[i] := FALSE;
    ReplaceList[i] := FALSE;
    LoadOnlyList[i] := FALSE;
    UnloadOnlyList[i] := FALSE;
    OldMatched[i] := FALSE;
    NewMatchedToOld[i] := 0;
    OldAssignedDrawer[i] := 0;
    OldAssignedSlot[i] := 0;
    NewAssignedDrawer[i] := 0;
    NewAssignedSlot[i] := 0;
    UnloadTempOrigIndex[i] := 0;
    UnloadTempDrawerIdx[i] := 0;
    UnloadTempOrder[i] := 0;
    LoadTempOrigIndex[i] := 0;
    LoadTempDrawerIdx[i] := 0;
    LoadTempOrder[i] := 0;
    KeepTempOrigIndex[i] := 0;
    KeepTempDrawerIdx[i] := 0;
    KeepTempOrder[i] := 0;
    OldAvailIndex[i] := 0;
    uniqName[i] := '';
    uniqLen[i] := '';
    uniqHorn[i] := '';
    uniqNeeded[i] := 0;
  END_FOR;

  FOR e := 1 TO 16 DO
    FOR d := 1 TO 15 DO
      usedSlot[e,d] := FALSE;
      usedSlotOld[e,d] := FALSE;
    END_FOR;
  END_FOR;

  FOR i := 1 TO 15 DO
    FOR j := 1 TO 12 DO
      Retain_UnloadPunchList[i,j] := '';
      Retain_ATCUnloadPunchList[i,j] := '';
      Retain_LoadPunchList[i,j] := '';
      Retain_ATCLoadPunchList[i,j] := '';
      Retain_KeepPunchList[i,j] := '';
      Retain_KeepPunchListNew[i,j] := '';
      Retain_RelocatePunchList[i,j] := '';
    END_FOR;
  END_FOR;

  RETURN;
END_IF;


(* Main execution: run on rising edge of Trigger or Unload *)
IF Unload THEN
  Busy := TRUE;
  Done := FALSE;
  Error := FALSE;
  ErrorText := '';
  ValidityFlag := TRUE;

  numSame := 0;
  numDiff := 0;
  newCount := 0;
  oldCount := 0;

  FOR i := 1 TO 15 DO
    KeepList[i] := FALSE;
    ReplaceList[i] := FALSE;
    LoadOnlyList[i] := FALSE;
    UnloadOnlyList[i] := FALSE;
    PerIndexStatus[i] := PIndexStatus.SAME;
    OldMatched[i] := FALSE;
    NewMatchedToOld[i] := 0;
    OldAssignedDrawer[i] := 0;
    OldAssignedSlot[i] := 0;
  END_FOR;

  FOR i := 1 TO 15 DO
    FOR j := 1 TO 12 DO
      Retain_UnloadPunchList[i,j] := '';
      Retain_ATCUnloadPunchList[i,j] := '';
      Retain_LoadPunchList[i,j] := '';
      Retain_ATCLoadPunchList[i,j] := '';
      Retain_KeepPunchList[i,j] := '';
      Retain_KeepPunchListNew[i,j] := '';
      Retain_RelocatePunchList[i,j] := '';
    END_FOR;
  END_FOR;

  FOR i := 1 TO 15 DO
    IF PFields_Old[i,3] <> '' THEN
      oldCount := oldCount + 1;
      UnloadOnlyList[i] := TRUE;
      PerIndexStatus[i] := PIndexStatus.DIFF;
    ELSE
      PerIndexStatus[i] := PIndexStatus.SAME;
    END_IF;
  END_FOR;

  (* --- Unload punchlari icin ATC drawer/slot cozumleme --- *)
  FOR e := 1 TO 16 DO
    FOR d := 1 TO 15 DO
      usedSlotOld[e,d] := FALSE;
    END_FOR;
  END_FOR;

  FOR i := 1 TO 15 DO
    IF UnloadOnlyList[i] THEN
      lenRealLocal := 0.0;
      hornRealLocal := 0.0;

      IF PFields_Old[i,7] <> '' THEN
        lenRealLocal := STRING_TO_REAL(PFields_Old[i,7]);
      END_IF;

      IF PFields_Old[i,9] <> '' THEN
        hornRealLocal := STRING_TO_REAL(PFields_Old[i,9]);
      END_IF;

      resolveFound := FALSE;

      FOR e := 1 TO 16 DO
        IF pDrawerName[e] = PFields_Old[i,3] THEN
          FOR d := 1 TO 15 DO
            IF
              PunchUsed[e,d] AND
              (NOT usedSlotOld[e,d]) AND
              (ABS(PunchLength[e,d] - lenRealLocal) <= eps) AND
              (ABS(PunchHorn[e,d] - hornRealLocal) <= eps)
            THEN
              usedSlotOld[e,d] := TRUE;
              OldAssignedDrawer[i] := e;
              OldAssignedSlot[i] := d;
              resolveFound := TRUE;
              EXIT;
            END_IF;
          END_FOR;

          IF resolveFound THEN
            EXIT;
          END_IF;
        END_IF;
      END_FOR;

      IF NOT resolveFound THEN
        ValidityFlag := FALSE;
        PackageState := PPackageState.INVALID;
        NextActionHint := PNextActionHint.INVALID_ABORT;
        Error := TRUE;
        ErrorText := CONCAT('INVALID: unload punch ATC slot NOT found FOR tool: ', PFields_Old[i,3]);
        Busy := FALSE;
        Done := TRUE;
        RETURN;
      END_IF;
    END_IF;
  END_FOR;

  UnloadTempCount := 0;

  FOR i := 1 TO 15 DO
    IF UnloadOnlyList[i] THEN
      UnloadTempCount := UnloadTempCount + 1;
      UnloadTempOrigIndex[UnloadTempCount] := i;
      UnloadTempDrawerIdx[UnloadTempCount] := OldAssignedDrawer[i];
      UnloadTempOrder[UnloadTempCount] := UnloadTempCount;
    END_IF;
  END_FOR;

  numSame := 0;
  numDiff := oldCount;
  newCount := 0;

  FOR a := 1 TO UnloadTempCount - 1 DO
    minIdx := a;

    FOR b := a + 1 TO UnloadTempCount DO
      IF UnloadTempDrawerIdx[b] < UnloadTempDrawerIdx[minIdx] THEN
        minIdx := b;
      ELSIF
        (UnloadTempDrawerIdx[b] = UnloadTempDrawerIdx[minIdx]) AND
        (UnloadTempOrder[b] < UnloadTempOrder[minIdx]) THEN
        minIdx := b;
      END_IF;
    END_FOR;

    IF minIdx <> a THEN
      tmpInt := UnloadTempOrigIndex[a];
      UnloadTempOrigIndex[a] := UnloadTempOrigIndex[minIdx];
      UnloadTempOrigIndex[minIdx] := tmpInt;

      tmpInt := UnloadTempDrawerIdx[a];
      UnloadTempDrawerIdx[a] := UnloadTempDrawerIdx[minIdx];
      UnloadTempDrawerIdx[minIdx] := tmpInt;

      tmpInt := UnloadTempOrder[a];
      UnloadTempOrder[a] := UnloadTempOrder[minIdx];
      UnloadTempOrder[minIdx] := tmpInt;
    END_IF;
  END_FOR;

  UnloadCount := 0;

  FOR a := 1 TO UnloadTempCount DO
    UnloadCount := UnloadCount + 1;
    i := UnloadTempOrigIndex[a];

    FOR j := 1 TO 12 DO
      Retain_UnloadPunchList[UnloadCount, j] := PFields_Old[i, j];
    END_FOR;

    Retain_UnloadPunchList[UnloadCount, 10] := INT_TO_STRING(OldAssignedDrawer[i]);
    Retain_UnloadPunchList[UnloadCount, 11] := INT_TO_STRING(OldAssignedSlot[i]);
    Retain_UnloadPunchList[UnloadCount, 12] := INT_TO_STRING(i);

    (* Drag / Direct: cozumlenen gercek drawer'a gore *)
    e := OldAssignedDrawer[i];

    IF (e = 4) OR (e = 8) OR (e = 12) OR (e = 16) THEN
      Retain_UnloadPunchList[UnloadCount,2] := '0';
    ELSE
      Retain_UnloadPunchList[UnloadCount,2] := '1';
    END_IF;
  END_FOR;

  FOR c := 1 TO 15 DO
    FOR j := 1 TO 12 DO
      Retain_ATCUnloadPunchList[c, j] := '';
    END_FOR;
  END_FOR;

  FOR c := 1 TO UnloadCount DO
    tmpInt := UnloadTempOrigIndex[c];
    e := OldAssignedDrawer[tmpInt];
    d := OldAssignedSlot[tmpInt];

    IF (e >= 1) AND (e <= 16) AND (d >= 1) AND (d <= 15) THEN
      Retain_ATCUnloadPunchList[c,1] := pDrawerName[e];
      Retain_ATCUnloadPunchList[c,2] := INT_TO_STRING(e);
      Retain_ATCUnloadPunchList[c,3] := INT_TO_STRING(d);
      Retain_ATCUnloadPunchList[c,4] := REAL_TO_STRING(PunchLength[e,d]);
      Retain_ATCUnloadPunchList[c,5] := REAL_TO_STRING(PunchPosition[e,d]);
      Retain_ATCUnloadPunchList[c,6] := REAL_TO_STRING(PunchHorn[e,d]);
      Retain_ATCUnloadPunchList[c,12] := INT_TO_STRING(tmpInt);
    END_IF;
  END_FOR;

  PackageState := PPackageState.PARTIAL_CHANGE;
  IF oldCount = 0 THEN
    NextActionHint := PNextActionHint.NO_ACTION;
  ELSE
    NextActionHint := PNextActionHint.ACTION_REQUIRED;
  END_IF;
  ValidityFlag := TRUE;

  Busy := FALSE;
  Done := TRUE;


ELSIF Trigger THEN
  FOR i := 1 TO 20 DO
    SPunchDrawerNo[i] := 0;
    SPunchIndex[i] := 0;
  END_FOR;

  Busy := TRUE;
  Done := FALSE;
  Error := FALSE;
  ErrorText := '';
  ValidityFlag := TRUE;

  numSame := 0;
  numDiff := 0;
  newCount := 0;
  oldCount := 0;

  FOR i := 1 TO 15 DO
    KeepList[i] := FALSE;
    ReplaceList[i] := FALSE;
    LoadOnlyList[i] := FALSE;
    UnloadOnlyList[i] := FALSE;
    PerIndexStatus[i] := PIndexStatus.SAME;
    OldMatched[i] := FALSE;
    NewMatchedToOld[i] := 0;
    OldAssignedDrawer[i] := 0;
    OldAssignedSlot[i] := 0;
    NewAssignedDrawer[i] := 0;
    NewAssignedSlot[i] := 0;
  END_FOR;

  FOR i := 1 TO 15 DO
    FOR j := 1 TO 12 DO
      Retain_UnloadPunchList[i,j] := '';
      Retain_ATCUnloadPunchList[i,j] := '';
      Retain_LoadPunchList[i,j] := '';
      Retain_ATCLoadPunchList[i,j] := '';
      Retain_KeepPunchList[i,j] := '';
      Retain_KeepPunchListNew[i,j] := '';
      Retain_RelocatePunchList[i,j] := '';
    END_FOR;
  END_FOR;

  (* STEP 1: validate new program *)
  FOR i := 1 TO 15 DO
    NewName := PFields[i,3];

    IF NewName <> '' THEN
      newCount := newCount + 1;
      NewLen := PFields[i,7];

      foundIndex := 0;

      FOR j := 1 TO 16 DO
        IF pDrawerName[j] = NewName THEN
          foundIndex := j;
          EXIT;
        END_IF;
      END_FOR;

      IF foundIndex = 0 THEN
        ValidityFlag := FALSE;
        PackageState := PPackageState.INVALID;
        NextActionHint := PNextActionHint.INVALID_ABORT;
        Error := TRUE;
        ErrorText := CONCAT('INVALID: missing ATC tool name: ', NewName);
        Busy := FALSE;
        Done := TRUE;
        RETURN;
      END_IF;

      IF NewLen = '' THEN
        ValidityFlag := FALSE;
        PackageState := PPackageState.INVALID;
        NextActionHint := PNextActionHint.INVALID_ABORT;
        Error := TRUE;
        ErrorText := CONCAT('INVALID: missing length for tool: ', NewName);
        Busy := FALSE;
        Done := TRUE;
        RETURN;
      END_IF;

      foundLengthMatch := FALSE;
      hasLengthValue := FALSE;
      NewLenReal := 0.0;

      IF NewLen <> '' THEN
        NewLenReal := STRING_TO_REAL(NewLen);
        hasLengthValue := TRUE;
      END_IF;

      IF hasLengthValue THEN
        FOR j := 1 TO 16 DO
          IF pDrawerName[j] = NewName THEN
            FOR k := 1 TO 15 DO
              IF ABS(PunchLength[j,k] - NewLenReal) <= eps THEN
                foundLengthMatch := TRUE;
                EXIT;
              END_IF;
            END_FOR;

            IF foundLengthMatch THEN
              EXIT;
            END_IF;
          END_IF;
        END_FOR;
      END_IF;

      IF NOT foundLengthMatch THEN
        ValidityFlag := FALSE;
        PackageState := PPackageState.INVALID;
        NextActionHint := PNextActionHint.INVALID_ABORT;
        Error := TRUE;
        ErrorText := CONCAT('INVALID: requested length NOT found FOR tool: ', NewName);
        Busy := FALSE;
        Done := TRUE;
        RETURN;
      END_IF;
    END_IF;
  END_FOR;


  (* STEP 2: keep matching *)
  OldAvailCount := 0;

  FOR j := 1 TO 15 DO
    OldName := PFields_Old[j,3];

    IF OldName <> '' THEN
      OldAvailCount := OldAvailCount + 1;
      OldAvailIndex[OldAvailCount] := j;
      OldMatched[j] := FALSE;
      oldCount := oldCount + 1;
    END_IF;
  END_FOR;

  FOR i := 1 TO 15 DO
    NewName := PFields[i,3];

    IF NewName <> '' THEN
      NewLen := PFields[i,7];
      NewHorn := PFields[i,9];
      NewRot := PFields[i,8];

      NewLenReal := 0.0;

      IF NewLen <> '' THEN
        NewLenReal := STRING_TO_REAL(NewLen);
      END_IF;

      NewMatchedToOld[i] := 0;
      a := 1;

      WHILE
        (a <= OldAvailCount) AND
        (NewMatchedToOld[i] = 0)
      DO
        j := OldAvailIndex[a];
        OldName := PFields_Old[j,3];

        IF OldName <> '' THEN
          OldLen := PFields_Old[j,7];
          OldHorn := PFields_Old[j,9];
          OldRot := PFields_Old[j,8];

          OldLenReal := 0.0;

          IF OldLen <> '' THEN
            OldLenReal := STRING_TO_REAL(OldLen);
          END_IF;

          IF
            (NewName = OldName) AND
            (NewHorn = OldHorn) AND
            (NewRot = OldRot) AND
            (ABS(NewLenReal - OldLenReal) <= eps)
          THEN
            OldMatched[j] := TRUE;
            KeepList[j] := TRUE;
            NewMatchedToOld[i] := j;

            FOR b := a TO OldAvailCount - 1 DO
              OldAvailIndex[b] := OldAvailIndex[b + 1];
            END_FOR;

            OldAvailCount := OldAvailCount - 1;
          ELSE
            a := a + 1;
          END_IF;
        ELSE
          a := a + 1;
        END_IF;
      END_WHILE;

      IF NewMatchedToOld[i] = 0 THEN
        LoadOnlyList[i] := TRUE;
      END_IF;
    END_IF;
  END_FOR;

  FOR a := 1 TO OldAvailCount DO
    j := OldAvailIndex[a];

    IF PFields_Old[j,3] <> '' THEN
      UnloadOnlyList[j] := TRUE;
    END_IF;
  END_FOR;


  (* STEP 3: validate load-only punches *)
  FOR i := 1 TO 15 DO
    IF LoadOnlyList[i] THEN
      NewName := PFields[i,3];
      NewLen := PFields[i,7];

      foundLengthMatch := FALSE;
      hasLengthValue := FALSE;
      NewLenReal := 0.0;

      IF NewLen <> '' THEN
        NewLenReal := STRING_TO_REAL(NewLen);
        hasLengthValue := TRUE;
      END_IF;

      IF hasLengthValue THEN
        FOR j := 1 TO 16 DO
          IF pDrawerName[j] = NewName THEN
            FOR k := 1 TO 15 DO
              IF
                (NOT PunchUsed[j,k]) AND
                (ABS(PunchLength[j,k] - NewLenReal) <= eps)
              THEN
                foundLengthMatch := TRUE;
                EXIT;
              END_IF;
            END_FOR;

            IF foundLengthMatch THEN
              EXIT;
            END_IF;
          END_IF;
        END_FOR;
      END_IF;

      IF NOT foundLengthMatch THEN
        ValidityFlag := FALSE;
        PackageState := PPackageState.INVALID;
        NextActionHint := PNextActionHint.INVALID_ABORT;
        Error := TRUE;
        ErrorText := CONCAT(
          'INVALID: requested load punch length NOT available in free ATC punches FOR tool: ',
          NewName
        );
        Busy := FALSE;
        Done := TRUE;
        RETURN;
      END_IF;
    END_IF;
  END_FOR;


  (* STEP 4: multiplicity check for load-only *)
  uniqCount := 0;

  FOR i := 1 TO 15 DO
    IF LoadOnlyList[i] THEN
      foundIndex := 0;

      FOR a := 1 TO uniqCount DO
        IF
          (uniqName[a] = PFields[i,3]) AND
          (uniqLen[a] = PFields[i,7]) AND
          (uniqHorn[a] = PFields[i,9])
        THEN
          foundIndex := a;
          EXIT;
        END_IF;
      END_FOR;

      IF foundIndex = 0 THEN
        uniqCount := uniqCount + 1;
        uniqName[uniqCount] := PFields[i,3];
        uniqLen[uniqCount] := PFields[i,7];
        uniqHorn[uniqCount] := PFields[i,9];
        uniqNeeded[uniqCount] := 1;
      ELSE
        uniqNeeded[foundIndex] := uniqNeeded[foundIndex] + 1;
      END_IF;
    END_IF;
  END_FOR;

  FOR a := 1 TO uniqCount DO
    NewName := uniqName[a];
    NewLen := uniqLen[a];
    NewHorn := uniqHorn[a];

    availCount := 0;

    IF (NewLen <> '') AND (NewHorn <> '') THEN
      NewLenReal := STRING_TO_REAL(NewLen);
      NewHornReal := STRING_TO_REAL(NewHorn);

      FOR e := 1 TO 16 DO
        IF pDrawerName[e] = NewName THEN
          FOR d := 1 TO 15 DO
            IF
              (NOT PunchUsed[e,d]) AND
              (ABS(PunchLength[e,d] - NewLenReal) <= eps) AND
              (ABS(PunchHorn[e,d] - NewHornReal) <= eps)
            THEN
              availCount := availCount + 1;
            END_IF;
          END_FOR;
        END_IF;
      END_FOR;
    END_IF;

    IF availCount < uniqNeeded[a] THEN
      ValidityFlag := FALSE;
      PackageState := PPackageState.INVALID;
      NextActionHint := PNextActionHint.INVALID_ABORT;
      Error := TRUE;
      ErrorText := CONCAT(
        'INVALID: insufficient FREE ATC slots for load tool ',
        NewName
      );
      Busy := FALSE;
      Done := TRUE;
      RETURN;
    END_IF;
  END_FOR;


  (* SUMMARY *)
  numSame := 0;
  numDiff := 0;

  FOR j := 1 TO 15 DO
    IF KeepList[j] THEN
      numSame := numSame + 1;
    END_IF;

    IF UnloadOnlyList[j] THEN
      numDiff := numDiff + 1;
    END_IF;
  END_FOR;

  FOR i := 1 TO 15 DO
    IF LoadOnlyList[i] THEN
      numDiff := numDiff + 1;
    END_IF;
  END_FOR;

  (* Compare tool and position pairs independently of keep pairing. *)
  noActionAllMatched := (newCount = oldCount);
  FOR j := 1 TO 15 DO
    noActionMatchedOld[j] := FALSE;
  END_FOR;

  FOR i := 1 TO 15 DO
    IF PFields[i,3] <> '' THEN
      noActionFound := FALSE;
      IF (PFields[i,6] <> '') AND (PFields[i,7] <> '') AND
         (PFields[i,8] <> '') AND (PFields[i,9] <> '') THEN
        FOR j := 1 TO 15 DO
          IF (NOT noActionMatchedOld[j]) AND
             (PFields[i,3] = PFields_Old[j,3]) AND
             (PFields[i,6] = PFields_Old[j,6]) AND
             (PFields[i,7] = PFields_Old[j,7]) AND
             (PFields[i,8] = PFields_Old[j,8]) AND
             (PFields[i,9] = PFields_Old[j,9]) THEN
            noActionMatchedOld[j] := TRUE;
            noActionFound := TRUE;
            EXIT;
          END_IF;
        END_FOR;
      END_IF;
      IF NOT noActionFound THEN
        noActionAllMatched := FALSE;
      END_IF;
    END_IF;
  END_FOR;

  IF NOT ValidityFlag THEN
    PackageState := PPackageState.INVALID;
    NextActionHint := PNextActionHint.INVALID_ABORT;
  ELSE
    PackageState := PPackageState.PARTIAL_CHANGE;
    IF noActionAllMatched THEN
      NextActionHint := PNextActionHint.NO_ACTION;
    ELSE
      NextActionHint := PNextActionHint.ACTION_REQUIRED;
    END_IF;
  END_IF;

  ValidityFlag := NOT Error;


  IF PackageState <> PPackageState.INVALID THEN

    FOR i := 1 TO 15 DO
      FOR j := 1 TO 12 DO
        Retain_UnloadPunchList[i,j] := '';
        Retain_ATCUnloadPunchList[i,j] := '';
        Retain_LoadPunchList[i,j] := '';
        Retain_ATCLoadPunchList[i,j] := '';
        Retain_KeepPunchList[i,j] := '';
        Retain_KeepPunchListNew[i,j] := '';
        Retain_RelocatePunchList[i,j] := '';
      END_FOR;
    END_FOR;


    (* ---------------------------------------------------------- *)
    (* Unload punchlari icin ATC drawer/slot cozumleme            *)
    (* PunchUsed=TRUE ve isim/boy/horn eslesen ilk slot secilir.  *)
    (* ---------------------------------------------------------- *)
    FOR e := 1 TO 16 DO
      FOR d := 1 TO 15 DO
        usedSlotOld[e,d] := FALSE;
      END_FOR;
    END_FOR;

    FOR i := 1 TO 15 DO
      IF UnloadOnlyList[i] THEN
        lenRealLocal := 0.0;
        hornRealLocal := 0.0;

        IF PFields_Old[i,7] <> '' THEN
          lenRealLocal := STRING_TO_REAL(PFields_Old[i,7]);
        END_IF;

        IF PFields_Old[i,9] <> '' THEN
          hornRealLocal := STRING_TO_REAL(PFields_Old[i,9]);
        END_IF;

        resolveFound := FALSE;

        FOR e := 1 TO 16 DO
          IF pDrawerName[e] = PFields_Old[i,3] THEN
            FOR d := 1 TO 15 DO
              IF
                PunchUsed[e,d] AND
                (NOT usedSlotOld[e,d]) AND
                (ABS(PunchLength[e,d] - lenRealLocal) <= eps) AND
                (ABS(PunchHorn[e,d] - hornRealLocal) <= eps)
              THEN
                usedSlotOld[e,d] := TRUE;
                OldAssignedDrawer[i] := e;
                OldAssignedSlot[i] := d;
                resolveFound := TRUE;
                EXIT;
              END_IF;
            END_FOR;

            IF resolveFound THEN
              EXIT;
            END_IF;
          END_IF;
        END_FOR;

        IF NOT resolveFound THEN
          ValidityFlag := FALSE;
          PackageState := PPackageState.INVALID;
          NextActionHint := PNextActionHint.INVALID_ABORT;
          Error := TRUE;
          ErrorText := CONCAT('INVALID: unload punch ATC slot NOT found FOR tool: ', PFields_Old[i,3]);
          Busy := FALSE;
          Done := TRUE;
          RETURN;
        END_IF;
      END_IF;
    END_FOR;


    (* pack unload *)
    UnloadTempCount := 0;

    FOR i := 1 TO 15 DO
      IF UnloadOnlyList[i] THEN
        UnloadTempCount := UnloadTempCount + 1;
        UnloadTempOrigIndex[UnloadTempCount] := i;
        UnloadTempDrawerIdx[UnloadTempCount] := OldAssignedDrawer[i];
        UnloadTempOrder[UnloadTempCount] := UnloadTempCount;
      END_IF;
    END_FOR;

    FOR a := 1 TO UnloadTempCount - 1 DO
      minIdx := a;

      FOR b := a + 1 TO UnloadTempCount DO
        IF UnloadTempDrawerIdx[b] < UnloadTempDrawerIdx[minIdx] THEN
          minIdx := b;
        ELSIF
          (UnloadTempDrawerIdx[b] = UnloadTempDrawerIdx[minIdx]) AND
          (UnloadTempOrder[b] < UnloadTempOrder[minIdx])
        THEN
          minIdx := b;
        END_IF;
      END_FOR;

      IF minIdx <> a THEN
        tmpInt := UnloadTempOrigIndex[a];
        UnloadTempOrigIndex[a] := UnloadTempOrigIndex[minIdx];
        UnloadTempOrigIndex[minIdx] := tmpInt;

        tmpInt := UnloadTempDrawerIdx[a];
        UnloadTempDrawerIdx[a] := UnloadTempDrawerIdx[minIdx];
        UnloadTempDrawerIdx[minIdx] := tmpInt;

        tmpInt := UnloadTempOrder[a];
        UnloadTempOrder[a] := UnloadTempOrder[minIdx];
        UnloadTempOrder[minIdx] := tmpInt;
      END_IF;
    END_FOR;

    UnloadCount := 0;

    FOR a := 1 TO UnloadTempCount DO
      UnloadCount := UnloadCount + 1;
      i := UnloadTempOrigIndex[a];

      FOR j := 1 TO 12 DO
        Retain_UnloadPunchList[UnloadCount,j] := PFields_Old[i,j];
      END_FOR;

      Retain_UnloadPunchList[UnloadCount,10] := INT_TO_STRING(OldAssignedDrawer[i]);
      Retain_UnloadPunchList[UnloadCount,11] := INT_TO_STRING(OldAssignedSlot[i]);
      Retain_UnloadPunchList[UnloadCount,12] := INT_TO_STRING(i);

      (* Drag / Direct: cozumlenen gercek drawer'a gore *)
      e := OldAssignedDrawer[i];

      IF (e = 4) OR (e = 8) OR (e = 12) OR (e = 16) THEN
        Retain_UnloadPunchList[UnloadCount,2] := '0';
      ELSE
        Retain_UnloadPunchList[UnloadCount,2] := '1';
      END_IF;
    END_FOR;

    (* ATC unload mapping *)
    FOR c := 1 TO UnloadCount DO
      tmpInt := UnloadTempOrigIndex[c];
      e := OldAssignedDrawer[tmpInt];
      d := OldAssignedSlot[tmpInt];

      IF (e >= 1) AND (e <= 16) AND (d >= 1) AND (d <= 15) THEN
        Retain_ATCUnloadPunchList[c,1] := pDrawerName[e];
        Retain_ATCUnloadPunchList[c,2] := INT_TO_STRING(e);
        Retain_ATCUnloadPunchList[c,3] := INT_TO_STRING(d);
        Retain_ATCUnloadPunchList[c,4] := REAL_TO_STRING(PunchLength[e,d]);
        Retain_ATCUnloadPunchList[c,5] := REAL_TO_STRING(PunchPosition[e,d]);
        Retain_ATCUnloadPunchList[c,6] := REAL_TO_STRING(PunchHorn[e,d]);
        Retain_ATCUnloadPunchList[c,12] := INT_TO_STRING(tmpInt);
      END_IF;
    END_FOR;


    (* pack load *)
    LoadTempCount := 0;

    FOR i := 1 TO 15 DO
      IF LoadOnlyList[i] THEN
        foundIndex := 0;

        FOR k := 1 TO 16 DO
          IF pDrawerName[k] = PFields[i,3] THEN
            foundIndex := k;
            EXIT;
          END_IF;
        END_FOR;

        IF foundIndex <> 0 THEN
          LoadTempCount := LoadTempCount + 1;
          LoadTempOrigIndex[LoadTempCount] := i;
          LoadTempDrawerIdx[LoadTempCount] := foundIndex;
          LoadTempOrder[LoadTempCount] := LoadTempCount;
        END_IF;
      END_IF;
    END_FOR;

    FOR a := 1 TO LoadTempCount - 1 DO
      minIdx := a;

      FOR b := a + 1 TO LoadTempCount DO
        IF LoadTempDrawerIdx[b] < LoadTempDrawerIdx[minIdx] THEN
          minIdx := b;
        ELSIF
          (LoadTempDrawerIdx[b] = LoadTempDrawerIdx[minIdx]) AND
          (LoadTempOrder[b] < LoadTempOrder[minIdx])
        THEN
          minIdx := b;
        END_IF;
      END_FOR;

      IF minIdx <> a THEN
        tmpInt := LoadTempOrigIndex[a];
        LoadTempOrigIndex[a] := LoadTempOrigIndex[minIdx];
        LoadTempOrigIndex[minIdx] := tmpInt;

        tmpInt := LoadTempDrawerIdx[a];
        LoadTempDrawerIdx[a] := LoadTempDrawerIdx[minIdx];
        LoadTempDrawerIdx[minIdx] := tmpInt;

        tmpInt := LoadTempOrder[a];
        LoadTempOrder[a] := LoadTempOrder[minIdx];
        LoadTempOrder[minIdx] := tmpInt;
      END_IF;
    END_FOR;

    LoadCount := 0;

    FOR a := 1 TO LoadTempCount DO
      LoadCount := LoadCount + 1;
      i := LoadTempOrigIndex[a];

      FOR j := 1 TO 12 DO
        Retain_LoadPunchList[LoadCount,j] := PFields[i,j];
      END_FOR;

      (* [10], [11] ve [2] (drag/direct) ATC eslestirmesinden sonra doldurulur *)
      Retain_LoadPunchList[LoadCount,2] := '';
      Retain_LoadPunchList[LoadCount,10] := '';
      Retain_LoadPunchList[LoadCount,11] := '';
      Retain_LoadPunchList[LoadCount,12] := INT_TO_STRING(i);
    END_FOR;


    (* ATC load mapping + drag/direct etiketleme *)
    FOR c := 1 TO 15 DO
      FOR j := 1 TO 12 DO
        Retain_ATCLoadPunchList[c,j] := '';
      END_FOR;
    END_FOR;

    FOR e := 1 TO 16 DO
      FOR d := 1 TO 15 DO
        usedSlot[e,d] := FALSE;
      END_FOR;
    END_FOR;

    FOR i := 1 TO 15 DO
      NewAssignedDrawer[i] := 0;
      NewAssignedSlot[i] := 0;
    END_FOR;

    FOR c := 1 TO LoadCount DO
      origNewIdx := LoadTempOrigIndex[c];
      NewName := Retain_LoadPunchList[c,3];
      lenStr := Retain_LoadPunchList[c,7];
      hornStr := Retain_LoadPunchList[c,9];
      foundIndex := 0;

      IF
        (NewName <> '') AND
        (lenStr <> '') AND
        (hornStr <> '')
      THEN
        lenRealLocal := STRING_TO_REAL(lenStr);
        hornRealLocal := STRING_TO_REAL(hornStr);

        FOR e := 1 TO 16 DO
          IF pDrawerName[e] = NewName THEN

            isDirectDrawer := (e = 4) OR (e = 8) OR (e = 12) OR (e = 16);

            (* direct drawer: kucukten buyuge, drag drawer: buyukten kucuge *)
            FOR m := 1 TO 15 DO
              IF isDirectDrawer THEN
                d := m;
              ELSE
                d := 16 - m;
              END_IF;

              IF
                (NOT usedSlot[e,d]) AND
                (NOT PunchUsed[e,d]) AND
                (ABS(PunchLength[e,d] - lenRealLocal) <= eps) AND
                (ABS(PunchHorn[e,d] - hornRealLocal) <= eps)
              THEN
                usedSlot[e,d] := TRUE;
                foundIndex := e;

                Retain_ATCLoadPunchList[c,1] := pDrawerName[e];
                Retain_ATCLoadPunchList[c,2] := INT_TO_STRING(e);   (* drawer no *)
                Retain_ATCLoadPunchList[c,3] := INT_TO_STRING(d);   (* slot no *)
                Retain_ATCLoadPunchList[c,4] := REAL_TO_STRING(PunchLength[e,d]);
                Retain_ATCLoadPunchList[c,5] := REAL_TO_STRING(PunchPosition[e,d]);
                Retain_ATCLoadPunchList[c,6] := REAL_TO_STRING(PunchHorn[e,d]);
                Retain_ATCLoadPunchList[c,12] := INT_TO_STRING(origNewIdx);

                NewAssignedDrawer[origNewIdx] := e;
                NewAssignedSlot[origNewIdx] := d;

                SPunchDrawerNo[origNewIdx] := e;
                SPunchIndex[origNewIdx] := d;

                Retain_LoadPunchList[c,10] := INT_TO_STRING(e);
                Retain_LoadPunchList[c,11] := INT_TO_STRING(d);

                IF isDirectDrawer THEN
                  Retain_LoadPunchList[c,2] := '0';
                ELSE
                  Retain_LoadPunchList[c,2] := '1';
                END_IF;

                EXIT;
              END_IF;
            END_FOR;

            IF foundIndex <> 0 THEN
              EXIT;
            END_IF;
          END_IF;
        END_FOR;
      END_IF;
    END_FOR;


    (* pack keep *)
    KeepTempCount := 0;

    FOR i := 1 TO 15 DO
      IF KeepList[i] THEN
        foundIndex := 0;

        FOR k := 1 TO 16 DO
          IF pDrawerName[k] = PFields_Old[i,3] THEN
            foundIndex := k;
            EXIT;
          END_IF;
        END_FOR;

        IF foundIndex <> 0 THEN
          KeepTempCount := KeepTempCount + 1;
          KeepTempOrigIndex[KeepTempCount] := i;
          KeepTempDrawerIdx[KeepTempCount] := foundIndex;
          KeepTempOrder[KeepTempCount] := KeepTempCount;
        END_IF;
      END_IF;
    END_FOR;

    FOR a := 1 TO KeepTempCount - 1 DO
      minIdx := a;

      FOR b := a + 1 TO KeepTempCount DO
        IF KeepTempDrawerIdx[b] < KeepTempDrawerIdx[minIdx] THEN
          minIdx := b;
        ELSIF
          (KeepTempDrawerIdx[b] = KeepTempDrawerIdx[minIdx]) AND
          (KeepTempOrder[b] < KeepTempOrder[minIdx])
        THEN
          minIdx := b;
        END_IF;
      END_FOR;

      IF minIdx <> a THEN
        tmpInt := KeepTempOrigIndex[a];
        KeepTempOrigIndex[a] := KeepTempOrigIndex[minIdx];
        KeepTempOrigIndex[minIdx] := tmpInt;

        tmpInt := KeepTempDrawerIdx[a];
        KeepTempDrawerIdx[a] := KeepTempDrawerIdx[minIdx];
        KeepTempDrawerIdx[minIdx] := tmpInt;

        tmpInt := KeepTempOrder[a];
        KeepTempOrder[a] := KeepTempOrder[minIdx];
        KeepTempOrder[minIdx] := tmpInt;
      END_IF;
    END_FOR;

    KeepCount := 0;

    FOR a := 1 TO KeepTempCount DO
      KeepCount := KeepCount + 1;
      i := KeepTempOrigIndex[a];

      FOR j := 1 TO 12 DO
        Retain_KeepPunchList[KeepCount,j] := PFields_Old[i,j];
        Retain_KeepPunchListNew[KeepCount,j] := '';
      END_FOR;

      Retain_KeepPunchList[KeepCount,12] := INT_TO_STRING(i);

      tmpInt := 0;

      FOR b := 1 TO 15 DO
        IF NewMatchedToOld[b] = i THEN
          tmpInt := b;
          EXIT;
        END_IF;
      END_FOR;

      IF tmpInt <> 0 THEN
        FOR j := 1 TO 12 DO
          Retain_KeepPunchListNew[KeepCount,j] := PFields[tmpInt,j];
        END_FOR;

        Retain_KeepPunchListNew[KeepCount,12] := INT_TO_STRING(tmpInt);

        (* Keep addresses come from the matched physical punch's old record. *)
        FOR indicatorColumn := 10 TO 11 DO
          indicatorValue := 0;
          indicatorValid := LEN(PFields_Old[i,indicatorColumn]) > 0;
          IF indicatorColumn = 10 THEN
            indicatorLimit := 16;
          ELSE
            indicatorLimit := 15;
          END_IF;

          FOR indicatorPos := 1 TO LEN(PFields_Old[i,indicatorColumn]) DO
            indicatorDigit := MID(PFields_Old[i,indicatorColumn], 1, indicatorPos);
            IF (indicatorDigit < '0') OR (indicatorDigit > '9') THEN
              indicatorValid := FALSE;
              EXIT;
            END_IF;

            indicatorValue := indicatorValue * 10 + STRING_TO_INT(indicatorDigit);
            IF indicatorValue > indicatorLimit THEN
              indicatorValid := FALSE;
              EXIT;
            END_IF;
          END_FOR;

          IF indicatorValid AND (indicatorValue > 0) THEN
            IF indicatorColumn = 10 THEN
              SPunchDrawerNo[tmpInt] := indicatorValue;
            ELSE
              SPunchIndex[tmpInt] := indicatorValue;
            END_IF;
          END_IF;
        END_FOR;
      END_IF;
    END_FOR;


    (* relocate threshold calculation *)
    (* Drag tespiti artik cozumlenen drawer'dan yapilir:         *)
    (*   unload -> OldAssignedDrawer, load -> NewAssignedDrawer  *)
    RightmostDragUnloadPos := -1.0;
    RightmostDragLoadPos := -1.0;

    FOR i := 1 TO 15 DO
      IF UnloadOnlyList[i] THEN
        eTmp := OldAssignedDrawer[i];

        IF
          (eTmp <> 4) AND
          (eTmp <> 8) AND
          (eTmp <> 12) AND
          (eTmp <> 16)
        THEN
          IF PFields_Old[i,6] > '' THEN
            tmpPos := STRING_TO_LREAL(PFields_Old[i,6]);

            IF tmpPos > RightmostDragUnloadPos THEN
              RightmostDragUnloadPos := tmpPos;
            END_IF;
          END_IF;
        END_IF;
      END_IF;
    END_FOR;

    FOR i := 1 TO 15 DO
      IF LoadOnlyList[i] THEN
        eTmp := NewAssignedDrawer[i];

        IF
          (eTmp <> 4) AND
          (eTmp <> 8) AND
          (eTmp <> 12) AND
          (eTmp <> 16)
        THEN
          IF PFields[i,6] > '' THEN
            tmpPos := STRING_TO_LREAL(PFields[i,6]);

            IF tmpPos > RightmostDragLoadPos THEN
              RightmostDragLoadPos := tmpPos;
            END_IF;
          END_IF;
        END_IF;
      END_IF;
    END_FOR;

    IF RightmostDragUnloadPos < 0.0 THEN
      RightmostDragUnloadPos := 0.0;
    END_IF;

    IF RightmostDragLoadPos < 0.0 THEN
      RightmostDragLoadPos := 0.0;
    END_IF;

    RelocateThreshold :=
      MAX(
        RightmostDragUnloadPos,
        RightmostDragLoadPos
      );


    (* ---------------------------------------------------------- *)
    (* Relocate list selection                                    *)
    (* ---------------------------------------------------------- *)
    RelocateCount := 0;

    FOR a := 1 TO KeepCount DO
      i := KeepTempOrigIndex[a];
      hasRelocateOverlap := FALSE;

      IF PFields_Old[i,6] > '' THEN
        tmpPos := STRING_TO_LREAL(PFields_Old[i,6]);

        IF tmpPos < RelocateThreshold THEN
          hasRelocateOverlap := TRUE;
        END_IF;
      END_IF;

      IF hasRelocateOverlap THEN
        RelocateCount := RelocateCount + 1;

        FOR j := 1 TO 12 DO
          Retain_RelocatePunchList[RelocateCount,j] :=
            Retain_KeepPunchList[a,j];
        END_FOR;
      END_IF;
    END_FOR;


    (* ---------------------------------------------------------- *)
    (* RelocateTrashold_Value refinement                          *)
    (* ---------------------------------------------------------- *)
    MaxKeepAboveThreshold := 0.0;
    FoundKeepAboveThreshold := FALSE;

    FOR a := 1 TO KeepCount DO
      IF Retain_KeepPunchList[a,6] <> '' THEN
        tmpPos := STRING_TO_LREAL(Retain_KeepPunchList[a,6]);

        IF tmpPos > RelocateThreshold THEN
          IF
            (NOT FoundKeepAboveThreshold) OR
            (tmpPos > MaxKeepAboveThreshold)
          THEN
            MaxKeepAboveThreshold := tmpPos;
            FoundKeepAboveThreshold := TRUE;
          END_IF;
        END_IF;
      END_IF;
    END_FOR;

    IF FoundKeepAboveThreshold THEN
      RelocateTrashold_Value := MaxKeepAboveThreshold;
    ELSE
      RelocateTrashold_Value := RelocateThreshold;
    END_IF;


    Busy := FALSE;
    Done := TRUE;

  ELSE
    Busy := FALSE;
    Done := TRUE;
  END_IF;

END_IF;
