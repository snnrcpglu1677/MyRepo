FUNCTION_BLOCK SKY_OPC_Comands
VAR_INPUT
    Prepare_Press : BOOL;
    Abort_Cycle : BOOL;

    Machine_StatesModes : DINT;                          // 1:Edit 2:Manual 3:Semi 4:Auto
    Status_Machine_HydraulicMotorRunningState : DINT;    // 1:Passive 2:Active
    Status_Machine_KeyRun : DINT;                        // 1:birak, 2:bas
    Status_Machine_KeyStop : DINT;                       // 1:birak, 2:bas
    Status_Machine_ExtraFunction : DINT;                 // 1:IndexPageOpen, 3:MenuPageOpened
    Status_Machine_Indexed : BOOL;
    Status_Machine_Clamped : BOOL;
    Status_Product_ProductName : STRING;
    Status_Machine_MovementStates : DINT;
    ATC_Cycle_Finish : BOOL;
END_VAR

VAR_OUTPUT
    Command_Machine_StatesModes : DINT;                  // 1:Edit 2:Manual 3:Semi 4:Auto
    Command_Machine_KeyRun : DINT;                       // 1:birak, 2:bas
    Command_Machine_KeyStop : DINT;                      // 1:birak, 2:bas
    Command_Machine_ExtraFunction : DINT;                // 1:IndexPageOpen, 3:MenuPageOpened
    Press_Ready : BOOL;
    Write_Product : BOOL;
    Rotation_Arm_Error : BOOL;
END_VAR

VAR
    CurrentState : Press_Prepare := Press_Prepare.PP_IDLE;
    lastPreparePress : BOOL := FALSE;
    preparePressRising : BOOL := FALSE;

    DownCommand_Delay : TON;
    AutoComand_Delay : TON;
    AutoComand_Delay_1 : TON;
    Product_Selection_Delay : TON;
    Product_Selection_Delay_1 : TON;
    Product_Clear_Delay : TON;
    Manual_Mode_Edit_Delay : TON;
    Manual_Mode_Auto_Delay : TON;
    Automode_Edit_Delay : TON;
    Automode_Auto_Delay : TON;
    Run_Key_Release_Again_Delay : TON;

    HM_Tolerance : REAL := 1;
END_VAR


preparePressRising := Prepare_Press AND NOT lastPreparePress;
DownCommand_Delay(PT := TO_TIME(2500));

Product_Clear_Delay(
    IN := NOT Abort_Cycle AND
          (CurrentState = Press_Prepare.PP_WAIT_PRODUCT_COMMAND_CLEARED),
    PT := TO_TIME(1000)
);

Product_Selection_Delay(
    IN := NOT Abort_Cycle AND
          (CurrentState = Press_Prepare.PP_WAIT_PRODUCT_SELECTED),
    PT := TO_TIME(1000)
);

Product_Selection_Delay_1(
    IN := NOT Abort_Cycle AND
          (CurrentState = Press_Prepare.PP_WAIT_SELECTED_PRODUCT_COMAND),
    PT := TO_TIME(1000)
);

Manual_Mode_Edit_Delay(
    IN := NOT Abort_Cycle AND
          (CurrentState = Press_Prepare.PP_WAIT_MANUAL_MODE) AND
          (Command_Machine_StatesModes = 1),
    PT := TO_TIME(1000)
);

Manual_Mode_Auto_Delay(
    IN := NOT Abort_Cycle AND
          (CurrentState = Press_Prepare.PP_WAIT_MANUAL_MODE) AND
          (Command_Machine_StatesModes = 4),
    PT := TO_TIME(1000)
);

Automode_Edit_Delay(
    IN := NOT Abort_Cycle AND
          (CurrentState = Press_Prepare.PP_WAIT_AUTOMODE) AND
          (Command_Machine_StatesModes = 1),
    PT := TO_TIME(1000)
);

Automode_Auto_Delay(
    IN := NOT Abort_Cycle AND
          (CurrentState = Press_Prepare.PP_WAIT_AUTOMODE) AND
          (Command_Machine_StatesModes = 4),
    PT := TO_TIME(1000)
);

AutoComand_Delay(
    IN := NOT Abort_Cycle AND
          (CurrentState = Press_Prepare.PP_WAIT_RUN_KEY_PRESS),
    PT := TO_TIME(1000)
);

AutoComand_Delay_1(
    IN := NOT Abort_Cycle AND
          (CurrentState = Press_Prepare.PP_WAIT_RUN_KEY_PRESS_AGAIN),
    PT := TO_TIME(1000)
);

Run_Key_Release_Again_Delay(
    IN := NOT Abort_Cycle AND
          (CurrentState = Press_Prepare.PP_WAIT_RUN_KEY_RELEASE_AGAIN),
    PT := TO_TIME(1000)
);


IF Abort_Cycle THEN

    CurrentState := Press_Prepare.PP_IDLE;
    Press_Ready := FALSE;
    Write_Product := FALSE;
    DownCommand_Delay.IN := FALSE;

    Command_Machine_StatesModes := 0;
    Command_Machine_KeyRun := 1;
    Command_Machine_KeyStop := 1;
    Command_Machine_ExtraFunction := 0;
    Command_Product_ProductName := '';

    // FMSC_PLC_IO.Akas_Piston_Up_Valve := FALSE;
    // FMSC_PLC_IO.Akas_Piston_Down_Valve := FALSE;
    FMSC_PLC_IO.Press_Wait_LDP_Comand := FALSE;
    FMSC_PLC_IO.Press_Down_Comand := FALSE;

ELSE

    CASE CurrentState OF

        Press_Prepare.PP_IDLE:
            Press_Ready := FALSE;
            Command_Machine_KeyRun := 1;
            Command_Machine_KeyStop := 1;

            IF preparePressRising AND
               (Status_Machine_HydraulicMotorRunningState = 2) AND
               (Status_Machine_Indexed = TRUE) THEN
                CurrentState := Press_Prepare.PP_START_PREPARE;
            END_IF;

        Press_Prepare.PP_START_PREPARE:
            IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - 135) <= HM_Tolerance THEN
                FMSC_PLC_IO.Akas_Piston_Up_Valve := TRUE;
                FMSC_PLC_IO.Akas_Piston_Down_Valve := FALSE;
                FMSC_PLC_IO.Press_Lower_ClampOn := TRUE;
                FMSC_PLC_IO.Press_Upper_ClampOn := TRUE;
                CurrentState := Press_Prepare.PP_CLEAR_PRODUCT_COMMAND;
            ELSE
                Rotation_Arm_Error := TRUE;
                CurrentState := Press_Prepare.PP_IDLE;
            END_IF;

        Press_Prepare.PP_CLEAR_PRODUCT_COMMAND:
            Command_Product_ProductName := Status_Product_ProductName;
            Write_Product := TRUE;
            CurrentState := Press_Prepare.PP_WAIT_PRODUCT_COMMAND_CLEARED;

        Press_Prepare.PP_WAIT_PRODUCT_COMMAND_CLEARED:
            IF Command_Product_ProductName = Status_Product_ProductName THEN
                CurrentState := Press_Prepare.PP_SET_PRODUCT_COMMAND;
                Write_Product := FALSE;
            ELSIF Product_Clear_Delay.Q THEN
                CurrentState := Press_Prepare.PP_CLEAR_PRODUCT_COMMAND;
            END_IF;

        Press_Prepare.PP_SET_PRODUCT_COMMAND:
            Command_Product_ProductName := 'ATC_Pre';
            Write_Product := TRUE;
            CurrentState := Press_Prepare.PP_WAIT_PRODUCT_SELECTED;

        Press_Prepare.PP_WAIT_PRODUCT_SELECTED:
            IF Product_Selection_Delay.Q THEN
                IF Status_Product_ProductName = Command_Product_ProductName THEN
                    Write_Product := FALSE;
                    CurrentState := Press_Prepare.PP_SET_MANUAL_MODE;
                ELSE
                    Command_Product_ProductName := Status_Product_ProductName;
                    CurrentState := Press_Prepare.PP_SET_PRODUCT_COMMAND;
                END_IF;
            END_IF;

        Press_Prepare.PP_SET_MANUAL_MODE:
            Command_Machine_StatesModes := 1;
            CurrentState := Press_Prepare.PP_WAIT_MANUAL_MODE;

        Press_Prepare.PP_WAIT_MANUAL_MODE:
            IF Machine_StatesModes = 1 THEN
                Command_Machine_StatesModes := 4;
            ELSIF Machine_StatesModes = 4 THEN
                CurrentState := Press_Prepare.PP_SET_PRESS_WAIT_LDP;
            ELSIF ((Command_Machine_StatesModes = 1) AND Manual_Mode_Edit_Delay.Q) OR
                  ((Command_Machine_StatesModes = 4) AND Manual_Mode_Auto_Delay.Q) THEN
                CurrentState := Press_Prepare.PP_SET_MANUAL_MODE;
            END_IF;

        Press_Prepare.PP_SET_PRESS_WAIT_LDP:
            FMSC_PLC_IO.Press_Wait_LDP_Comand := TRUE;
            CurrentState := Press_Prepare.PP_RUN_KEY_PRESS;

        Press_Prepare.PP_RUN_KEY_PRESS:
            Command_Machine_KeyRun := 2;
            CurrentState := Press_Prepare.PP_WAIT_RUN_KEY_PRESS;

        Press_Prepare.PP_WAIT_RUN_KEY_PRESS:
            IF (Status_Machine_KeyRun = 2) AND AutoComand_Delay.Q THEN
                CurrentState := Press_Prepare.PP_RUN_KEY_RELEASE;
            END_IF;

        Press_Prepare.PP_RUN_KEY_RELEASE:
            Command_Machine_KeyRun := 1;
            CurrentState := Press_Prepare.PP_WAIT_RUN_KEY_RELEASE;

        Press_Prepare.PP_WAIT_RUN_KEY_RELEASE:
            // Mevcut 2 kontrolu korundu; test sonrasi 1 olarak degerlendirilecek.
            IF Status_Machine_KeyRun = 2 THEN
                CurrentState := Press_Prepare.PP_SET_PRESS_DOWN;
            END_IF;

        Press_Prepare.PP_SET_PRESS_DOWN:
            IF FMSC_PLC_IO.Press_Lower_Clamp_Ok AND
               FMSC_PLC_IO.Press_Upper_Clamp_Ok AND
               (Machine_StatesModes = 4) THEN
                DownCommand_Delay.IN := TRUE;
                IF DownCommand_Delay.Q THEN
                    FMSC_PLC_IO.Press_Down_Comand := TRUE;
                    CurrentState := Press_Prepare.PP_WAIT_PRESS_DOWN_COMPLETE;
                END_IF;
            END_IF;

        Press_Prepare.PP_WAIT_PRESS_DOWN_COMPLETE:
            DownCommand_Delay.IN := FALSE;
            IF Status_Machine_MovementStates = 4 THEN
                CurrentState := Press_Prepare.PP_SET_AKAS_PISTON_UP;
            END_IF;

        Press_Prepare.PP_SET_AKAS_PISTON_UP:
            FMSC_PLC_IO.Akas_Piston_Up_Valve := TRUE;
            FMSC_PLC_IO.Akas_Piston_Down_Valve := FALSE;
            FMSC_PLC_IO.Press_Lower_ClampOn := FALSE;
            FMSC_PLC_IO.Press_Upper_ClampOn := FALSE;
            CurrentState := Press_Prepare.PP_WAIT_AKAS_PISTON_UP_POSITION;

        Press_Prepare.PP_WAIT_AKAS_PISTON_UP_POSITION:
            IF Fmsc_PLC_IO.Akas_Piston_Up_Position AND
               NOT FMSC_PLC_IO.Press_Lower_Clamp_Ok AND
               NOT FMSC_PLC_IO.Press_Upper_Clamp_Ok THEN
                CurrentState := Press_Prepare.PP_WAIT_ATC_CYCLE_FINISH;
            END_IF;

        Press_Prepare.PP_WAIT_ATC_CYCLE_FINISH:
            Press_Ready := TRUE;
            IF ATC_Cycle_Finish THEN
                FMSC_PLC_IO.Press_Wait_LDP_Comand := FALSE;
                FMSC_PLC_IO.Press_Down_Comand := FALSE;
                IF FMSC_PLC_IO.Press_At_TDC THEN
                    CurrentState := Press_Prepare.PP_SET_SELECTED_PRODUCT_COMAND;
                END_IF;
            END_IF;

        Press_Prepare.PP_SET_SELECTED_PRODUCT_COMAND:
            Command_Product_ProductName := Product_Name;
            Write_Product := TRUE;
            CurrentState := Press_Prepare.PP_WAIT_SELECTED_PRODUCT_COMAND;

        Press_Prepare.PP_WAIT_SELECTED_PRODUCT_COMAND:
            IF Product_Selection_Delay_1.Q THEN
                IF Status_Product_ProductName = Command_Product_ProductName THEN
                    Write_Product := FALSE;
                    CurrentState := Press_Prepare.PP_SET_AUTOMODE;
                ELSE
                    Command_Product_ProductName := Status_Product_ProductName;
                    CurrentState := Press_Prepare.PP_SET_SELECTED_PRODUCT_COMAND;
                END_IF;
            END_IF;

        Press_Prepare.PP_SET_AUTOMODE:
            Command_Machine_StatesModes := 1;
            CurrentState := Press_Prepare.PP_WAIT_AUTOMODE;

        Press_Prepare.PP_WAIT_AUTOMODE:
            IF Command_Machine_StatesModes = 1 THEN
                IF Machine_StatesModes = 1 THEN
                    Command_Machine_StatesModes := 4;
                ELSIF Automode_Edit_Delay.Q THEN
                    CurrentState := Press_Prepare.PP_SET_AUTOMODE;
                END_IF;
            ELSIF Command_Machine_StatesModes = 4 THEN
                IF Machine_StatesModes = 4 THEN
                    CurrentState := Press_Prepare.PP_RUN_KEY_PRESS_AGAIN;
                ELSIF Automode_Auto_Delay.Q THEN
                    CurrentState := Press_Prepare.PP_SET_AUTOMODE;
                END_IF;
            END_IF;

        Press_Prepare.PP_RUN_KEY_PRESS_AGAIN:
            Command_Machine_KeyRun := 2;
            CurrentState := Press_Prepare.PP_WAIT_RUN_KEY_PRESS_AGAIN;

        Press_Prepare.PP_WAIT_RUN_KEY_PRESS_AGAIN:
            IF (Status_Machine_KeyRun = 2) AND AutoComand_Delay_1.Q THEN
                CurrentState := Press_Prepare.PP_RUN_KEY_RELEASE_AGAIN;
            END_IF;

        Press_Prepare.PP_RUN_KEY_RELEASE_AGAIN:
            Command_Machine_KeyRun := 1;
            CurrentState := Press_Prepare.PP_WAIT_RUN_KEY_RELEASE_AGAIN;

        Press_Prepare.PP_WAIT_RUN_KEY_RELEASE_AGAIN:
            IF Status_Machine_KeyRun = 1 THEN
                CurrentState := Press_Prepare.PP_DONE;
            ELSIF Run_Key_Release_Again_Delay.Q THEN
                CurrentState := Press_Prepare.PP_WAIT_RUN_KEY_PRESS_AGAIN;
            END_IF;

        Press_Prepare.PP_DONE:
            Press_Ready := FALSE;
            IF NOT Prepare_Press THEN
                CurrentState := Press_Prepare.PP_IDLE;
            END_IF;

    END_CASE;

END_IF;

lastPreparePress := Prepare_Press;
