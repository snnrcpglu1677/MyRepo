FUNCTION_BLOCK Punch_DraggUnload

VAR_INPUT
  UnloadStartCmd      : BOOL;    (* State_Machine'den gelen başlat komutu *)
  Reset               : BOOL;    (* SOFT reset: state/substate resetlemez, sadece requestleri keser *)
  Abort_Cycle         : BOOL;    (* HARD abort: tüm FB içi state/substate/flagleri resetler *)
  Automatic_Page      : BOOL;    (* TRUE: otomatik sayfa aktif, FALSE: otomatik dışı -> pause zorunlu *)
END_VAR

VAR_IN_OUT
  General_Stop        : BOOL;
  General_Start       : BOOL;    (* Enable/pause: FALSE iken pause, TRUE iken devam *)
END_VAR

VAR_OUTPUT
  UnloadStartAck      : BOOL;    (* State_Machine'e ACK *)
  UnloadBusy          : BOOL;    (* FB aktif çalışıyor *)
  UnloadError         : BOOL;    (* hata flag *)
  UnloadDone          : BOOL;    (* tüm unload bitti *)
END_VAR

VAR RETAIN
  {attribute 'ac_persist':='PersistenceChannel'} Local_Shuttle_State     : INT := 0;
  {attribute 'ac_persist':='PersistenceChannel'} Local_WareHouse_State   : INT := 0;
  {attribute 'ac_persist':='PersistenceChannel'} Local_RotArm_State      : INT := 0;

  {attribute 'ac_persist':='PersistenceChannel'} UnloadPunchCount        : INT := 0;
  {attribute 'ac_persist':='PersistenceChannel'} DragPunchCount          : INT := 0;

  {attribute 'ac_persist':='PersistenceChannel'} PDraggUnload_WH_SubStates     : INT := 0;
  {attribute 'ac_persist':='PersistenceChannel'} PDraggUnload_SH_SubStates     : INT := 0;
  {attribute 'ac_persist':='PersistenceChannel'} PDraggUnload_RotArm_SubStates : INT := 0;

  lastStartCmd          : BOOL := FALSE;
  initiated             : BOOL := FALSE;

  Current_PunchIndex    : INT := 0;
  Current_PunchLoaded   : BOOL := FALSE;
  LastInitializedPunch  : INT := 0;

  (* Drag ordering helper arrays *)
  DragSrcIndex          : ARRAY[1..15] OF INT;
  DragDrawerNo          : ARRAY[1..15] OF INT;
  DragSlotNo            : ARRAY[1..15] OF INT;
  DragOrderUsed         : ARRAY[1..15] OF BOOL;

  (* Warehouse helper flags *)
  WH_TargetDrawerIdx    : INT := 0;
  WH_TargetRackID       : INT := 0;
  WH_NeedRackBring      : BOOL := FALSE;
  WH_NeedRackChange     : BOOL := FALSE;
  WH_SameRack           : BOOL := FALSE;

  WH_R_MoveRequested    : BOOL := FALSE;
  WH_X_MoveRequested    : BOOL := FALSE;
  WH_R_Done             : BOOL := FALSE;
  WH_X_Done             : BOOL := FALSE;
  WH_WaitRackInPos      : BOOL := FALSE;

  WH_Eps                : REAL := 0.01;

  WH_R_LoadRequested    : BOOL := FALSE;
  WH_R_LoadDone         : BOOL := FALSE;
  R_Not_Safety_Zone     : BOOL := FALSE;
  Rack_Presence_Error   : BOOL := FALSE;

  (* Shuttle helper flags *)
  SH_DX_MoveRequested   : BOOL := FALSE;
  SH_DX_Done            : BOOL := FALSE;
  SH_DR_MoveRequested   : BOOL := FALSE;
  SH_DR_Done            : BOOL := FALSE;
  SH_Z_MoveRequested    : BOOL := FALSE;
  SH_Z_Done             : BOOL := FALSE;

  (* Rotation arm helper flags *)
  RA_W_MoveRequested    : BOOL := FALSE;
  RA_W_Done             : BOOL := FALSE;
  RotArm_ReadyForUnload : BOOL := FALSE;
  RotArm_RotationDone   : BOOL := FALSE;
  RotArm_PunchReleased  : BOOL := FALSE;
  RotRequired           : BOOL := FALSE;

  DragPrepared          : BOOL := FALSE;
END_VAR

VAR
  risingStart           : BOOL;
  AutoRunEnable         : BOOL;
  i                     : INT;
  j                     : INT;
  k                     : INT;
  tmpInt                : INT;
  tmpInt2               : INT;
  tmpDrawer             : INT;
  tmpSlot               : INT;
  tmpWidth              : LREAL;
  tmpHalfWidth          : LREAL;
  tmpCurrentPos         : LREAL;
  tmpSrc                : INT;
  tmpOldIndex           : INT;

  PosEps                : LREAL := 0.01;
  KeepPosEps            : LREAL := 0.01;

  ZOffset               : LREAL;
  TargetCenter          : LREAL;

  ClampOff_Timer        : TON;
  ClampOn_Timer         : TON;
END_VAR


(* ------------------------------------------------------------ *)
(* Combined enable *)
(* ------------------------------------------------------------ *)
AutoRunEnable := (General_Start AND Automatic_Page);

ClampOff_Timer(PT := TO_TIME(TO_INT(Retain_GeneralParameters[4] * 1000)));
ClampOn_Timer(PT := TO_TIME(TO_INT(Retain_GeneralParameters[3] * 1000)));

risingStart := (UnloadStartCmd AND NOT lastStartCmd);
ZOffset := REAL_TO_LREAL(Retain_ZParameters[19]);


(* ------------------------------------------------------------ *)
(* HARD ABORT *)
(* ------------------------------------------------------------ *)
IF Abort_Cycle THEN
  UnloadStartAck := FALSE;
  UnloadBusy := FALSE;
  UnloadError := FALSE;
  UnloadDone := FALSE;

  lastStartCmd := FALSE;
  initiated := FALSE;
  DragPrepared := FALSE;

  Current_PunchIndex := 0;
  Current_PunchLoaded := FALSE;
  LastInitializedPunch := 0;

  UnloadPunchCount := 0;
  DragPunchCount := 0;

  Local_Shuttle_State := 0;
  Local_WareHouse_State := 0;
  Local_RotArm_State := 0;

  PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_IDLE;
  PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.SH_IDLE;
  PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_IDLE;

  WH_TargetDrawerIdx := 0;
  WH_TargetRackID := 0;
  WH_NeedRackBring := FALSE;
  WH_NeedRackChange := FALSE;
  WH_SameRack := FALSE;

  WH_R_MoveRequested := FALSE;
  WH_X_MoveRequested := FALSE;
  WH_R_Done := FALSE;
  WH_X_Done := FALSE;
  WH_WaitRackInPos := FALSE;

  WH_R_LoadRequested := FALSE;
  WH_R_LoadDone := FALSE;
  R_Not_Safety_Zone := FALSE;
  Rack_Presence_Error := FALSE;

  SH_DX_MoveRequested := FALSE;
  SH_DX_Done := FALSE;
  SH_DR_MoveRequested := FALSE;
  SH_DR_Done := FALSE;
  SH_Z_MoveRequested := FALSE;
  SH_Z_Done := FALSE;

  RA_W_MoveRequested := FALSE;
  RA_W_Done := FALSE;
  RotArm_ReadyForUnload := FALSE;
  RotArm_RotationDone := FALSE;
  RotArm_PunchReleased := FALSE;
  RotRequired := FALSE;

  arAxes[EN_Axes.ATC_DR].AxisAutoSetVelocity := Retain_DRParameters[1];

  arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_DX].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_DR].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_Z].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := FALSE;

  RETURN;
END_IF


(* ------------------------------------------------------------ *)
(* PAUSE *)
(* ------------------------------------------------------------ *)
IF NOT AutoRunEnable AND NOT PLC_PRG.HMI.Drawer_Demand_Enable THEN
  arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_DX].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_DR].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_Z].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := FALSE;

  lastStartCmd := UnloadStartCmd;
  RETURN;
END_IF


(* ------------------------------------------------------------ *)
(* SOFT RESET *)
(* ------------------------------------------------------------ *)
IF Reset THEN
  arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_DX].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_DR].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_Z].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := FALSE;
END_IF


(* ------------------------------------------------------------ *)
(* START + DRAG LIST PREPARATION *)
(* ------------------------------------------------------------ *)
IF risingStart THEN
  initiated := TRUE;
  UnloadBusy := TRUE;
  UnloadError := FALSE;
  UnloadDone := FALSE;
  DragPrepared := FALSE;

  UnloadPunchCount := 0;
  DragPunchCount := 0;

  FOR i := 1 TO 15 DO
    DragSrcIndex[i] := 0;
    DragDrawerNo[i] := 0;
    DragSlotNo[i] := 0;
    DragOrderUsed[i] := FALSE;
  END_FOR;

  (* Count all unload candidates *)
  FOR i := 1 TO 15 DO
    IF (Retain_UnloadPunchList[i,3] <> '') THEN
      UnloadPunchCount := UnloadPunchCount + 1;
    END_IF;
  END_FOR;

  (* Build drag source list from unload list *)
  FOR i := 1 TO 15 DO
    IF (Retain_UnloadPunchList[i,3] <> '') AND (Retain_UnloadPunchList[i,2] = '1') THEN
      DragPunchCount := DragPunchCount + 1;
      DragSrcIndex[DragPunchCount] := i;

      IF Retain_UnloadPunchList[i,12] <> '' THEN
        DragDrawerNo[DragPunchCount] := STRING_TO_INT(Retain_UnloadPunchList[i,12]);
      ELSE
        DragDrawerNo[DragPunchCount] := 0;
      END_IF;

      IF Retain_UnloadPunchList[i,11] <> '' THEN
        DragSlotNo[DragPunchCount] := STRING_TO_INT(Retain_UnloadPunchList[i,11]);
      ELSE
        DragSlotNo[DragPunchCount] := 0;
      END_IF;
    END_IF;
  END_FOR;

  (* Sort by drawer no, then slot no ascending *)
  FOR i := 1 TO DragPunchCount - 1 DO
    FOR j := i + 1 TO DragPunchCount DO
      IF (DragDrawerNo[j] < DragDrawerNo[i]) OR
         ((DragDrawerNo[j] = DragDrawerNo[i]) AND (DragSlotNo[j] < DragSlotNo[i])) THEN

        tmpInt := DragSrcIndex[i];
        DragSrcIndex[i] := DragSrcIndex[j];
        DragSrcIndex[j] := tmpInt;

        tmpInt := DragDrawerNo[i];
        DragDrawerNo[i] := DragDrawerNo[j];
        DragDrawerNo[j] := tmpInt;

        tmpInt := DragSlotNo[i];
        DragSlotNo[i] := DragSlotNo[j];
        DragSlotNo[j] := tmpInt;
      END_IF;
    END_IF;
  END_FOR;

  (* Copy to drag lists *)
  FOR i := 1 TO DragPunchCount DO
    tmpSrc := DragSrcIndex[i];

    FOR k := 1 TO 12 DO
      Retain_DragPunchList[i,k] := Retain_UnloadPunchList[tmpSrc,k];
      Retain_ATCDragPunchList[i,k] := Retain_ATCUnloadPunchList[tmpSrc,k];
    END_FOR;
  END_FOR;

  IF DragPunchCount = 0 THEN
    Current_PunchIndex := 0;
    Current_PunchLoaded := FALSE;
    UnloadDone := TRUE;
    initiated := FALSE;

    Local_WareHouse_State := 0;
    Local_Shuttle_State := 0;
    Local_RotArm_State := 0;

    PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_IDLE;
    PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.SH_IDLE;
    PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_IDLE;
  ELSE
    Current_PunchIndex := 1;
    Current_PunchLoaded := FALSE;
    UnloadDone := FALSE;
    DragPrepared := TRUE;

    Local_Shuttle_State := Shuttle_States.PRG_Proccesing;
    Local_WareHouse_State := WareHouse_States.PRG_Proccesing;
    Local_RotArm_State := RotationArm_States.U_RotArm_Init_And_Home;

    PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U26_INIT;
    PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_INIT;
    PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_CHECK_POSITION;
  END_IF;

  UnloadStartAck := TRUE;
END_IF


(* ------------------------------------------------------------ *)
(* ENGINE FLOWS *)
(* ------------------------------------------------------------ *)
IF initiated AND DragPrepared THEN

  (* ============================= *)
  (* ROTATION ARM ENGINE           *)
  (* ============================= *)
  IF UnloadStartCmd THEN
    CASE Local_RotArm_State OF

      RotationArm_States.U_RotArm_Init_And_Home:
        CASE PDraggUnload_RotArm_SubStates OF

          PDraggUnload_RotArm_SubState.ROT_IDLE:
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_CHECK_POSITION;

          PDraggUnload_RotArm_SubState.ROT_INIT_CHECK_POSITION:
            RotArm_ReadyForUnload := FALSE;
            RotArm_RotationDone := FALSE;
            RotArm_PunchReleased := FALSE;
            RA_W_MoveRequested := FALSE;
            RA_W_Done := FALSE;

            IF Fmsc_PLC_IO.Rotation_Arm_Inside THEN
              IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - 90.0) <= WH_Eps THEN
                PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_MOVE_OUTSIDE_CMD;
              ELSE
                PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_W_TO_90_REQ;
              END_IF;

            ELSIF Fmsc_PLC_IO.Rotation_Arm_Outside THEN
              IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - 180.0) <= WH_Eps THEN
                PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_LOCK_CMD;
              ELSE
                PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_W_TO_180_REQ;
              END_IF;

            ELSE
              UnloadError := TRUE;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INIT_W_TO_90_REQ:
            arAxes[EN_Axes.ATC_W].TargetPos := 90.0;
            arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := TRUE;
            RA_W_MoveRequested := TRUE;
            RA_W_Done := FALSE;
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_W_TO_90_WAIT;

          PDraggUnload_RotArm_SubState.ROT_INIT_W_TO_90_WAIT:
            IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - arAxes[EN_Axes.ATC_W].TargetPos) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := FALSE;
              RA_W_MoveRequested := FALSE;
              RA_W_Done := TRUE;
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_MOVE_OUTSIDE_CMD;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INIT_MOVE_OUTSIDE_CMD:
            Fmsc_PLC_IO.Rotation_Arm_Move_Inside_Valve := FALSE;
            Fmsc_PLC_IO.Rotation_Arm_Move_Outside_Valve := TRUE;
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_WAIT_OUTSIDE;

          PDraggUnload_RotArm_SubState.ROT_INIT_WAIT_OUTSIDE:
            IF Fmsc_PLC_IO.Rotation_Arm_Outside THEN
              Fmsc_PLC_IO.Rotation_Arm_Move_Outside_Valve := FALSE;

              IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - 180.0) <= WH_Eps THEN
                PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_LOCK_CMD;
              ELSE
                PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_W_TO_180_REQ;
              END_IF;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INIT_W_TO_180_REQ:
            arAxes[EN_Axes.ATC_W].TargetPos := 180.0;
            arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := TRUE;
            RA_W_MoveRequested := TRUE;
            RA_W_Done := FALSE;
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_W_TO_180_WAIT;

          PDraggUnload_RotArm_SubState.ROT_INIT_W_TO_180_WAIT:
            IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - arAxes[EN_Axes.ATC_W].TargetPos) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := FALSE;
              RA_W_MoveRequested := FALSE;
              RA_W_Done := TRUE;
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_LOCK_CMD;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INIT_LOCK_CMD:
            Fmsc_PLC_IO.Rotation_Arm_Unlocking_Valve := FALSE;
            Fmsc_PLC_IO.Rotation_Arm_Locking_Valve := TRUE;
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_WAIT_LOCKED;

          PDraggUnload_RotArm_SubState.ROT_INIT_WAIT_LOCKED:
            IF Fmsc_PLC_IO.Rotation_Arm_Locked_CH2 AND Fmsc_PLC_IO.Rotation_Arm_Locked_CH3 THEN
              RotArm_ReadyForUnload := TRUE;
              Local_RotArm_State := RotationArm_States.U_RotArm_Ready_For_Unload;
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_READY;
            END_IF;

        ELSE
          PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_CHECK_POSITION;
        END_CASE;

      RotationArm_States.U_RotArm_Ready_For_Unload:
        CASE PDraggUnload_RotArm_SubStates OF

          PDraggUnload_RotArm_SubState.ROT_READY:
            RotArm_ReadyForUnload := TRUE;
            IF RotRequired THEN
              Local_RotArm_State := RotationArm_States.U_RotArm_Rotation_Interrupt_Active;
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_WAIT_PUNCH_ON_ARM;
            END_IF;

        ELSE
          PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_READY;
        END_CASE;

      RotationArm_States.U_RotArm_Rotation_Interrupt_Active:
        CASE PDraggUnload_RotArm_SubStates OF

          PDraggUnload_RotArm_SubState.ROT_INT_WAIT_PUNCH_ON_ARM:
            RotArm_ReadyForUnload := FALSE;
            RotArm_RotationDone := FALSE;
            RotArm_PunchReleased := FALSE;
            IF Fmsc_PLC_IO.Punch_Locked_on_Rotation_Arm THEN
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_WAIT_SHUTTLE_CLEAR;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INT_WAIT_SHUTTLE_CLEAR:
            IF (PDraggUnload_SH_SubStates = PDraggUnload_SH_SubState.PUNLDRG_U28_DONE) THEN
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_ARM_UNLOCK_CMD;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INT_ARM_UNLOCK_CMD:
            Fmsc_PLC_IO.Rotation_Arm_Unlocking_Valve := TRUE;
            Fmsc_PLC_IO.Rotation_Arm_Locking_Valve := FALSE;
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_WAIT_ARM_UNLOCKED;

          PDraggUnload_RotArm_SubState.ROT_INT_WAIT_ARM_UNLOCKED:
            IF Fmsc_PLC_IO.Rotation_Arm_Unlocked_CH2 AND Fmsc_PLC_IO.Rotation_Arm_Unlocked_CH3 THEN
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_W_TO_ROTATED_REQ;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INT_W_TO_ROTATED_REQ:
            arAxes[EN_Axes.ATC_W].TargetPos := 0.0;
            arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := TRUE;
            RA_W_MoveRequested := TRUE;
            RA_W_Done := FALSE;
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_W_TO_ROTATED_WAIT;

          PDraggUnload_RotArm_SubState.ROT_INT_W_TO_ROTATED_WAIT:
            IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - arAxes[EN_Axes.ATC_W].TargetPos) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := FALSE;
              RA_W_MoveRequested := FALSE;
              RA_W_Done := TRUE;
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_ARM_LOCK_CMD;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INT_ARM_LOCK_CMD:
            Fmsc_PLC_IO.Rotation_Arm_Locking_Valve := TRUE;
            Fmsc_PLC_IO.Rotation_Arm_Unlocking_Valve := FALSE;
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_WAIT_ARM_LOCKED;

          PDraggUnload_RotArm_SubState.ROT_INT_WAIT_ARM_LOCKED:
            IF Fmsc_PLC_IO.Rotation_Arm_Locked_CH2 AND Fmsc_PLC_IO.Rotation_Arm_Locked_CH3 THEN
              RotArm_RotationDone := TRUE;
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_WAIT_SHUTTLE_PICKBACK;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INT_WAIT_SHUTTLE_PICKBACK:
            IF (PDraggUnload_SH_SubStates = PDraggUnload_SH_SubState.PUNLDRG_U30_WAIT_ROTARM_RELEASED) THEN
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_UNLOCK_CMD;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INT_UNLOCK_CMD:
            Fmsc_PLC_IO.Rotation_Arm_Unlocking_Valve := TRUE;
            Fmsc_PLC_IO.Rotation_Arm_Locking_Valve := FALSE;
            Fmsc_PLC_IO.Rotation_Arm_Punch_Lock_Valve := FALSE;
            Fmsc_PLC_IO.Rotation_Arm_Punch_Unlock_Valve := TRUE;
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_WAIT_PUNCH_RELEASED;

          PDraggUnload_RotArm_SubState.ROT_INT_WAIT_PUNCH_RELEASED:
            IF NOT Fmsc_PLC_IO.Punch_Locked_on_Rotation_Arm THEN
              RotArm_PunchReleased := TRUE;
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_RETURN_UNLOCK_CMD;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INT_RETURN_UNLOCK_CMD:
            Fmsc_PLC_IO.Rotation_Arm_Unlocking_Valve := TRUE;
            Fmsc_PLC_IO.Rotation_Arm_Locking_Valve := FALSE;
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_RETURN_WAIT_UNLOCKED;

          PDraggUnload_RotArm_SubState.ROT_INT_RETURN_WAIT_UNLOCKED:
            IF Fmsc_PLC_IO.Rotation_Arm_Unlocked_CH2 AND Fmsc_PLC_IO.Rotation_Arm_Unlocked_CH3 THEN
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_RETURN_W_TO_180_REQ;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INT_RETURN_W_TO_180_REQ:
            IF ABS(arAxes[EN_Axes.ATC_Z].ActualPos <= TO_LREAL(Retain_ZParameters[8])) THEN
              arAxes[EN_Axes.ATC_W].TargetPos := 180.0;
              arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := TRUE;
              RA_W_MoveRequested := TRUE;
              RA_W_Done := FALSE;
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_RETURN_W_TO_180_WAIT;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INT_RETURN_W_TO_180_WAIT:
            IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - arAxes[EN_Axes.ATC_W].TargetPos) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := FALSE;
              RA_W_MoveRequested := FALSE;
              RA_W_Done := TRUE;
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_RETURN_LOCK_CMD;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INT_RETURN_LOCK_CMD:
            Fmsc_PLC_IO.Rotation_Arm_Locking_Valve := TRUE;
            Fmsc_PLC_IO.Rotation_Arm_Unlocking_Valve := FALSE;
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_RETURN_WAIT_LOCKED;

          PDraggUnload_RotArm_SubState.ROT_INT_RETURN_WAIT_LOCKED:
            IF Fmsc_PLC_IO.Rotation_Arm_Locked_CH2 AND Fmsc_PLC_IO.Rotation_Arm_Locked_CH3 THEN
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_DONE;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INT_DONE:
            RotRequired := FALSE;
            RotArm_ReadyForUnload := TRUE;
            Local_RotArm_State := RotationArm_States.U_RotArm_Ready_For_Unload;
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_READY;

        ELSE
          PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_WAIT_PUNCH_ON_ARM;
        END_CASE;

    ELSE
      ;
    END_CASE;
  END_IF;


  (* ============================= *)
  (* WAREHOUSE ENGINE               *)
  (* ============================= *)
  IF (Local_WareHouse_State = WareHouse_States.PRG_Proccesing) THEN

    IF Current_PunchIndex = 0 THEN
      UnloadError := TRUE;
    ELSE
      IF NOT ((Retain_DragPunchList[Current_PunchIndex,3] <> '') AND (Retain_DragPunchList[Current_PunchIndex,2] = '1')) THEN
        Current_PunchIndex := Current_PunchIndex + 1;
      ELSE

        CASE PDraggUnload_WH_SubStates OF

          PDraggUnload_WH_SubState.PUNLDRG_IDLE:
            PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_INIT;

          PDraggUnload_WH_SubState.PUNLDRG_U14_INIT:
            WH_TargetDrawerIdx := 0;
            WH_TargetRackID := 0;
            WH_NeedRackBring := FALSE;
            WH_NeedRackChange := FALSE;
            WH_SameRack := FALSE;
            PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_READ_TARGETS;

          PDraggUnload_WH_SubState.PUNLDRG_U14_READ_TARGETS:
            IF Retain_ATCDragPunchList[Current_PunchIndex,2] <> '' THEN
              WH_TargetDrawerIdx := STRING_TO_INT(Retain_ATCDragPunchList[Current_PunchIndex,2]);
            ELSE
              WH_TargetDrawerIdx := 0;
            END_IF;

            IF WH_TargetDrawerIdx >= 1 THEN
              tmpInt := WH_TargetDrawerIdx - 1;
              WH_TargetRackID := TO_INT(tmpInt / 4) + 1;
            ELSE
              WH_TargetRackID := 0;
            END_IF;

            IF Current_Rack_Type = 0 THEN
              WH_NeedRackBring := TRUE;
              WH_NeedRackChange := FALSE;
              WH_SameRack := FALSE;
            ELSIF Current_Rack_Type = 1 THEN
              WH_NeedRackBring := FALSE;
              IF TO_INT(Current_Rack_ID) = WH_TargetRackID THEN
                WH_SameRack := TRUE;
                WH_NeedRackChange := FALSE;
              ELSE
                WH_SameRack := FALSE;
                WH_NeedRackChange := TRUE;
              END_IF;
            ELSIF Current_Rack_Type = 2 THEN
              WH_NeedRackBring := FALSE;
              WH_NeedRackChange := TRUE;
              WH_SameRack := FALSE;
            END_IF;

            IF WH_NeedRackBring THEN
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_UNLOCK_CHECK;
            ELSIF WH_SameRack THEN
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_SAMERACK_DRAWER_CHECK;
            ELSIF WH_NeedRackChange THEN
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_RACKCHANGE_UNLOCK_CHECK;
            ELSE
              UnloadError := TRUE;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U14_SAMERACK_DRAWER_CHECK:
            IF Current_Drawer_ID = WH_TargetDrawerIdx THEN
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_DONE;
            ELSE
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_SAMERACK_R_TO_DRAWER_REQ;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U14_SAMERACK_R_TO_DRAWER_REQ:
            IF (WH_TargetDrawerIdx >= 1) AND (WH_TargetDrawerIdx <= 16) THEN
              IF ABS(arAxes[EN_Axes.ATC_R].ActualPos - TO_LREAL(pDrawerParam1[WH_TargetDrawerIdx])) <= WH_Eps THEN
                arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
                WH_R_MoveRequested := FALSE;
                WH_R_Done := TRUE;
                Current_Drawer_ID := WH_TargetDrawerIdx;
                PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_DONE;
              ELSE
                arAxes[EN_Axes.ATC_R].TargetPos := TO_LREAL(pDrawerParam1[WH_TargetDrawerIdx]);
                arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := TRUE;
                WH_R_MoveRequested := TRUE;
                WH_R_Done := FALSE;
                PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_SAMERACK_R_TO_DRAWER_WAIT;
              END_IF;
            ELSE
              UnloadError := TRUE;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U14_SAMERACK_R_TO_DRAWER_WAIT:
            IF ABS(arAxes[EN_Axes.ATC_R].ActualPos - arAxes[EN_Axes.ATC_R].TargetPos) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
              WH_R_MoveRequested := FALSE;
              WH_R_Done := TRUE;
              Current_Drawer_ID := WH_TargetDrawerIdx;
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_DONE;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U14_RACKCHANGE_UNLOCK_CHECK:
            IF NOT (Fmsc_PLC_IO.Carriage_Unlocked_Left AND Fmsc_PLC_IO.Carriage_Unlocked_Right) THEN
              Fmsc_PLC_IO.Carriage_UnLocking_Valve := TRUE;
              FMSC_PLC_IO.Carriage_Locking_Valve := FALSE;
            ELSE
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_RACKCHANGE_R_TO_SAFE_REQ;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U14_RACKCHANGE_R_TO_SAFE_REQ:
            arAxes[EN_Axes.ATC_R].TargetPos := TO_LREAL(Retain_RParameters[3]);
            arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := TRUE;
            WH_R_MoveRequested := TRUE;
            WH_R_Done := FALSE;
            PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_RACKCHANGE_R_TO_SAFE_WAIT;

          PDraggUnload_WH_SubState.PUNLDRG_U14_RACKCHANGE_R_TO_SAFE_WAIT:
            IF ABS(arAxes[EN_Axes.ATC_R].ActualPos - arAxes[EN_Axes.ATC_R].TargetPos) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
              WH_R_MoveRequested := FALSE;
              WH_R_Done := TRUE;
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_RACKCHANGE_X_TO_CURRENT_DRAWER_REQ;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U14_RACKCHANGE_X_TO_CURRENT_DRAWER_REQ:
            IF (Current_Drawer_ID >= 1) AND (Current_Drawer_ID <= 16) THEN
              IF ABS(arAxes[EN_Axes.ATC_X].ActualPos - TO_LREAL(pDrawerParam2[TO_INT(Current_Drawer_ID)])) <= WH_Eps THEN
                arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := FALSE;
                WH_X_MoveRequested := FALSE;
                WH_X_Done := TRUE;
                PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_RACKCHANGE_R_TO_DROP_REQ;
              ELSE
                arAxes[EN_Axes.ATC_X].TargetPos := TO_LREAL(pDrawerParam2[TO_INT(Current_Drawer_ID)]);
                arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := TRUE;
                WH_X_MoveRequested := TRUE;
                WH_X_Done := FALSE;
                PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_RACKCHANGE_X_TO_CURRENT_DRAWER_WAIT;
              END_IF;
            ELSE
              UnloadError := TRUE;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U14_RACKCHANGE_X_TO_CURRENT_DRAWER_WAIT:
            IF ABS(arAxes[EN_Axes.ATC_X].ActualPos - arAxes[EN_Axes.ATC_X].TargetPos) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := FALSE;
              WH_X_MoveRequested := FALSE;
              WH_X_Done := TRUE;
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_RACKCHANGE_R_TO_DROP_REQ;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U14_RACKCHANGE_R_TO_DROP_REQ:
            arAxes[EN_Axes.ATC_R].TargetPos := -10.0;
            arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := TRUE;
            WH_R_MoveRequested := TRUE;
            WH_R_Done := FALSE;
            PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_RACKCHANGE_R_TO_DROP_WAIT;

          PDraggUnload_WH_SubState.PUNLDRG_U14_RACKCHANGE_R_TO_DROP_WAIT:
            IF ABS(arAxes[EN_Axes.ATC_R].ActualPos - arAxes[EN_Axes.ATC_R].TargetPos) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
              WH_R_MoveRequested := FALSE;
              WH_R_Done := TRUE;
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_RACKCHANGE_WAIT_RACK_RELEASED;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U14_RACKCHANGE_WAIT_RACK_RELEASED:
            IF NOT Fmsc_PLC_IO.Rack_Presence THEN
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_RACKCHANGE_CLEAR_CURRENT_INFO;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U14_RACKCHANGE_CLEAR_CURRENT_INFO:
            Current_Rack_ID := 0;
            Current_Rack_Type := 0;
            Current_Drawer_ID := 0;
            WH_NeedRackChange := FALSE;
            WH_NeedRackBring := TRUE;
            WH_SameRack := FALSE;
            PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_UNLOCK_CHECK;

          PDraggUnload_WH_SubState.PUNLDRG_U14_UNLOCK_CHECK:
            IF NOT (Fmsc_PLC_IO.Carriage_Unlocked_Left AND Fmsc_PLC_IO.Carriage_Unlocked_Right) THEN
              Fmsc_PLC_IO.Carriage_UnLocking_Valve := TRUE;
            ELSE
              Fmsc_PLC_IO.Carriage_UnLocking_Valve := FALSE;
              IF R_Not_Safety_Zone THEN
                PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_WAIT_R_SAFEZONE;
              ELSE
                PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_R_TO_PICKUP_REQ;
              END_IF;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U14_WAIT_R_SAFEZONE:
            IF arAxes[EN_Axes.ATC_R].ActualPos < TO_LREAL(Retain_RParameters[7]) THEN
              R_Not_Safety_Zone := FALSE;
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_R_TO_PICKUP_REQ;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U14_R_TO_PICKUP_REQ:
            IF arAxes[EN_Axes.ATC_R].ActualPos < TO_LREAL(Retain_RParameters[7]) THEN
              arAxes[EN_Axes.ATC_R].TargetPos := -10.0;
              arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := TRUE;
              WH_R_MoveRequested := TRUE;
              WH_R_Done := FALSE;
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_R_TO_PICKUP_WAIT;
            ELSE
              R_Not_Safety_Zone := TRUE;
              WH_R_MoveRequested := FALSE;
              WH_R_Done := FALSE;
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_WAIT_R_SAFEZONE;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U14_R_TO_PICKUP_WAIT:
            IF ABS(arAxes[EN_Axes.ATC_R].ActualPos - (-10.0)) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
              WH_R_MoveRequested := FALSE;
              WH_R_Done := TRUE;
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_X_TO_DRAWER_REQ;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U14_X_TO_DRAWER_REQ:
            IF WH_TargetDrawerIdx >= 1 THEN
              IF ABS(arAxes[EN_Axes.ATC_X].ActualPos - TO_LREAL(pDrawerParam2[WH_TargetDrawerIdx])) <= WH_Eps THEN
                WH_X_Done := TRUE;
                WH_X_MoveRequested := FALSE;
                PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_R_LIFT_REQ;
              ELSE
                arAxes[EN_Axes.ATC_X].TargetPos := TO_LREAL(pDrawerParam2[WH_TargetDrawerIdx]);
                arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := TRUE;
                WH_X_MoveRequested := TRUE;
                WH_X_Done := FALSE;
                PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_X_TO_DRAWER_WAIT;
              END_IF;
            ELSE
              UnloadError := TRUE;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U14_X_TO_DRAWER_WAIT:
            IF ABS(arAxes[EN_Axes.ATC_X].ActualPos - arAxes[EN_Axes.ATC_X].TargetPos) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := FALSE;
              WH_X_MoveRequested := FALSE;
              WH_X_Done := TRUE;
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_R_LIFT_REQ;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U14_R_LIFT_REQ:
            arAxes[EN_Axes.ATC_R].TargetPos := TO_LREAL(Retain_RParameters[3]);
            arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := TRUE;
            WH_R_LoadRequested := TRUE;
            WH_R_LoadDone := FALSE;
            PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_R_LIFT_WAIT;

          PDraggUnload_WH_SubState.PUNLDRG_U14_R_LIFT_WAIT:
            IF arAxes[EN_Axes.ATC_R].ActualPos >= TO_LREAL(Retain_RParameters[7]) THEN
              IF Fmsc_PLC_IO.Rack_Presence THEN
                Current_Rack_ID := WH_TargetRackID;
                Current_Rack_Type := 1;
                Current_Drawer_ID := WH_TargetDrawerIdx;
              ELSE
                Rack_Presence_Error := TRUE;
                General_Stop := TRUE;
                General_Start := FALSE;
                PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_ERROR_RACK_PRESENCE;
              END_IF;
            END_IF;

            IF ABS(arAxes[EN_Axes.ATC_R].ActualPos - arAxes[EN_Axes.ATC_R].TargetPos) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
              WH_R_LoadRequested := FALSE;
              WH_R_LoadDone := TRUE;
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_WAIT_RACK_IN_POS;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U14_WAIT_RACK_IN_POS:
            WH_WaitRackInPos := (WH_R_LoadDone AND WH_X_Done);
            IF WH_WaitRackInPos THEN
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_DONE;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U14_ERROR_RACK_PRESENCE:
            IF arAxes[EN_Axes.ATC_R].ActualPos < TO_LREAL(Retain_RParameters[7]) THEN
              Rack_Presence_Error := FALSE;
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_R_LIFT_REQ;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U14_DONE:
            Local_WareHouse_State := WareHouse_States.U_Punches_rack_taken;
            PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U15_INIT;

          PDraggUnload_WH_SubState.PUNLDRG_U15_INIT:
            WH_X_MoveRequested := FALSE;
            WH_X_Done := FALSE;
            WH_R_MoveRequested := FALSE;
            WH_R_Done := FALSE;
            PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U15_X_TO_UNLPOS_REQ;

          PDraggUnload_WH_SubState.PUNLDRG_U15_X_TO_UNLPOS_REQ:
            IF ABS(arAxes[EN_Axes.ATC_X].ActualPos - TO_LREAL(Retain_XParameters[2])) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := FALSE;
              WH_X_MoveRequested := FALSE;
              WH_X_Done := TRUE;
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U15_R_TO_UNLPOS_REQ;
            ELSE
              arAxes[EN_Axes.ATC_X].TargetPos := TO_LREAL(Retain_XParameters[2]);
              arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := TRUE;
              WH_X_MoveRequested := TRUE;
              WH_X_Done := FALSE;
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U15_X_TO_UNLPOS_WAIT;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U15_X_TO_UNLPOS_WAIT:
            IF ABS(arAxes[EN_Axes.ATC_X].ActualPos - arAxes[EN_Axes.ATC_X].TargetPos) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := FALSE;
              WH_X_MoveRequested := FALSE;
              WH_X_Done := TRUE;
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U15_R_TO_UNLPOS_REQ;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U15_R_TO_UNLPOS_REQ:
            IF (WH_TargetDrawerIdx >= 1) AND (WH_TargetDrawerIdx <= 16) THEN
              IF ABS(arAxes[EN_Axes.ATC_R].ActualPos - TO_LREAL(pDrawerParam1[WH_TargetDrawerIdx])) <= WH_Eps THEN
                arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
                WH_R_MoveRequested := FALSE;
                WH_R_Done := TRUE;
                PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U15_DONE;
              ELSE
                arAxes[EN_Axes.ATC_R].TargetPos := TO_LREAL(pDrawerParam1[WH_TargetDrawerIdx]);
                arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := TRUE;
                WH_R_MoveRequested := TRUE;
                WH_R_Done := FALSE;
                PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U15_R_TO_UNLPOS_WAIT;
              END_IF;
            ELSE
              UnloadError := TRUE;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U15_R_TO_UNLPOS_WAIT:
            IF ABS(arAxes[EN_Axes.ATC_R].ActualPos - arAxes[EN_Axes.ATC_R].TargetPos) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
              WH_R_MoveRequested := FALSE;
              WH_R_Done := TRUE;
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U15_DONE;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U15_DONE:
            Local_WareHouse_State := WareHouse_States.U_PDrawer_Unload_pos_reached;
            PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U16_INIT;

          PDraggUnload_WH_SubState.PUNLDRG_U16_INIT:
            WH_X_MoveRequested := FALSE;
            WH_X_Done := FALSE;
            WH_R_MoveRequested := FALSE;
            WH_R_Done := FALSE;
            PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U16_LOCK_CMD;

          PDraggUnload_WH_SubState.PUNLDRG_U16_LOCK_CMD:
            Fmsc_PLC_IO.Carriage_UnLocking_Valve := FALSE;
            Fmsc_PLC_IO.Carriage_Locking_Valve := TRUE;
            PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U16_WAIT_LOCKED;

          PDraggUnload_WH_SubState.PUNLDRG_U16_WAIT_LOCKED:
            IF (Fmsc_PLC_IO.Carriage_Locked_Left AND Fmsc_PLC_IO.Carriage_Locked_Right) THEN
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U16_DONE;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U16_DONE:
            Local_WareHouse_State := WareHouse_States.U_Punch_Unload_drawer_allowed;
            PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U17_INIT;

          PDraggUnload_WH_SubState.PUNLDRG_U17_INIT:
            WH_X_MoveRequested := FALSE;
            WH_X_Done := FALSE;
            WH_R_MoveRequested := FALSE;
            WH_R_Done := FALSE;
            PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U17_SET_CURRENT_RACK;

          PDraggUnload_WH_SubState.PUNLDRG_U17_SET_CURRENT_RACK:
            PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U17_VERIFY_CURRENT_RACK;

          PDraggUnload_WH_SubState.PUNLDRG_U17_VERIFY_CURRENT_RACK:
            IF (TO_INT(Current_Rack_ID) = WH_TargetRackID) AND
               (Current_Rack_Type = 1) AND
               (Current_Drawer_ID = WH_TargetDrawerIdx) THEN
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U17_DONE;
            END_IF;

          PDraggUnload_WH_SubState.PUNLDRG_U17_DONE:
            PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U17_DONE;

        ELSE
          PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_INIT;
        END_CASE;
      END_IF;
    END_IF;
  END_IF;


  (* ============================= *)
  (* SHUTTLE ENGINE                 *)
  (* ============================= *)
  IF UnloadStartCmd THEN
    CASE Local_Shuttle_State OF

      Shuttle_States.PRG_Proccesing:
        CASE PDraggUnload_SH_SubStates OF

          PDraggUnload_SH_SubState.SH_IDLE:
            ;

          PDraggUnload_SH_SubState.PUNLDRG_U26_INIT:
            SH_DX_MoveRequested := FALSE;
            SH_DX_Done := FALSE;
            SH_DR_MoveRequested := FALSE;
            SH_DR_Done := FALSE;
            SH_Z_MoveRequested := FALSE;
            SH_Z_Done := FALSE;

              arAxes[EN_Axes.ATC_DR].TargetPos := TO_LREAL(pDrawerParam3[tmpInt]);

              arAxes[EN_Axes.ATC_Z].AxisTravelRequestAbs := TRUE;
              arAxes[EN_Axes.ATC_DR].AxisTravelRequestAbs := TRUE;

              SH_Z_MoveRequested := TRUE;
              SH_Z_Done := FALSE;
              SH_DR_MoveRequested := TRUE;
              SH_DR_Done := FALSE;

              PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U30_DR_Z_TO_ROT_PICKUP_WAIT;
            ELSE
              UnloadError := TRUE;
            END_IF;

          PDraggUnload_SH_SubState.PUNLDRG_U30_DR_Z_TO_ROT_PICKUP_WAIT:
            IF ABS(arAxes[EN_Axes.ATC_Z].ActualPos - arAxes[EN_Axes.ATC_Z].TargetPos) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_Z].AxisTravelRequestAbs := FALSE;
              SH_Z_MoveRequested := FALSE;
              SH_Z_Done := TRUE;
            END_IF;

            IF ABS(arAxes[EN_Axes.ATC_DR].ActualPos - arAxes[EN_Axes.ATC_DR].TargetPos) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_DR].AxisTravelRequestAbs := FALSE;
              SH_DR_MoveRequested := FALSE;
              SH_DR_Done := TRUE;
            END_IF;

            IF SH_DR_Done AND SH_Z_Done THEN
              PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U30_DX_TO_ROT_PICKUP_REQ;
            END_IF;

          PDraggUnload_SH_SubState.PUNLDRG_U30_DX_TO_ROT_PICKUP_REQ:
            IF Retain_ATCDragPunchList[Current_PunchIndex,2] <> '' THEN
              tmpInt := STRING_TO_INT(Retain_ATCDragPunchList[Current_PunchIndex,2]);
              arAxes[EN_Axes.ATC_DX].TargetPos := TO_LREAL(pDrawerParam7[tmpInt]);
              arAxes[EN_Axes.ATC_DX].AxisTravelRequestAbs := TRUE;
              SH_DX_MoveRequested := TRUE;
              SH_DX_Done := FALSE;
              PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U30_DX_TO_ROT_PICKUP_WAIT;
            ELSE
              UnloadError := TRUE;
            END_IF;

          PDraggUnload_SH_SubState.PUNLDRG_U30_DX_TO_ROT_PICKUP_WAIT:
            IF ABS(arAxes[EN_Axes.ATC_DX].ActualPos - arAxes[EN_Axes.ATC_DX].TargetPos) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_DX].AxisTravelRequestAbs := FALSE;
              SH_DX_MoveRequested := FALSE;
              SH_DX_Done := TRUE;
              PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U30_PIN1_OUTSIDE_CMD;
            END_IF;

          PDraggUnload_SH_SubState.PUNLDRG_U30_PIN1_OUTSIDE_CMD:
            Fmsc_PLC_IO.Shuttle_Pin1_Move_Inside_Valve := FALSE;
            Fmsc_PLC_IO.Shuttle_Pin1_Move_Outside_Valve := TRUE;
            PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U30_WAIT_PIN1_OUTSIDE;

          PDraggUnload_SH_SubState.PUNLDRG_U30_WAIT_PIN1_OUTSIDE:
            IF Fmsc_PLC_IO.Shuttle_Pin1_Outside THEN
              Fmsc_PLC_IO.Shuttle_Pin1_Move_Outside_Valve := FALSE;
              PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U30_PIN2_OUTSIDE_CMD;
            END_IF;

          PDraggUnload_SH_SubState.PUNLDRG_U30_PIN2_OUTSIDE_CMD:
            Fmsc_PLC_IO.Shuttle_Pin2_Move_Inside_Valve := FALSE;
            Fmsc_PLC_IO.Shuttle_Pin2_Move_Outside_Valve := TRUE;
            PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U30_WAIT_PIN2_OUTSIDE;

          PDraggUnload_SH_SubState.PUNLDRG_U30_WAIT_PIN2_OUTSIDE:
            IF Fmsc_PLC_IO.Shuttle_Pin2_Outside THEN
              Fmsc_PLC_IO.Shuttle_Pin2_Move_Outside_Valve := FALSE;
              PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U30_WAIT_ROTARM_RELEASED;
            END_IF;

          PDraggUnload_SH_SubState.PUNLDRG_U30_WAIT_ROTARM_RELEASED:
            IF RotArm_PunchReleased THEN
              PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U31_INIT;
            END_IF;

          PDraggUnload_SH_SubState.PUNLDRG_U31_INIT:
            SH_Z_MoveRequested := FALSE;
            SH_Z_Done := FALSE;
            SH_DR_MoveRequested := FALSE;
            SH_DR_Done := FALSE;
            PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U31_Z_DR_TO_WH_REPOS_REQ;

          PDraggUnload_SH_SubState.PUNLDRG_U31_Z_DR_TO_WH_REPOS_REQ:
            IF (Retain_ATCDragPunchList[Current_PunchIndex,2] <> '') AND
               (Retain_ATCDragPunchList[Current_PunchIndex,5] <> '') THEN
              tmpInt := STRING_TO_INT(Retain_ATCDragPunchList[Current_PunchIndex,2]);
              arAxes[EN_Axes.ATC_Z].TargetPos := STRING_TO_LREAL(Retain_ATCDragPunchList[Current_PunchIndex,5]);
              arAxes[EN_Axes.ATC_DR].TargetPos := TO_LREAL(pDrawerParam3[tmpInt]);

              arAxes[EN_Axes.ATC_Z].AxisTravelRequestAbs := TRUE;
              arAxes[EN_Axes.ATC_DR].AxisTravelRequestAbs := TRUE;

              SH_Z_MoveRequested := TRUE;
              SH_Z_Done := FALSE;
              SH_DR_MoveRequested := TRUE;
              SH_DR_Done := FALSE;

              PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U31_Z_DR_TO_WH_REPOS_WAIT;
            ELSE
              UnloadError := TRUE;
            END_IF;

          PDraggUnload_SH_SubState.PUNLDRG_U31_Z_DR_TO_WH_REPOS_WAIT:
            IF ABS(arAxes[EN_Axes.ATC_Z].ActualPos - arAxes[EN_Axes.ATC_Z].TargetPos) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_Z].AxisTravelRequestAbs := FALSE;
              SH_Z_MoveRequested := FALSE;
              SH_Z_Done := TRUE;
            END_IF;

            IF ABS(arAxes[EN_Axes.ATC_DR].ActualPos - arAxes[EN_Axes.ATC_DR].TargetPos) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_DR].AxisTravelRequestAbs := FALSE;
              SH_DR_MoveRequested := FALSE;
              SH_DR_Done := TRUE;
            END_IF;

            IF SH_Z_Done AND SH_DR_Done THEN
              PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U31_DONE;
            END_IF;

          PDraggUnload_SH_SubState.PUNLDRG_U31_DONE:
            Local_Shuttle_State := Shuttle_States.U_Punch_correctly_extracted;
            PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U32_INIT;

          PDraggUnload_SH_SubState.PUNLDRG_U32_INIT:
            SH_DX_MoveRequested := FALSE;
            SH_DX_Done := FALSE;
            PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U32_WAIT_WH_READY;

          PDraggUnload_SH_SubState.PUNLDRG_U32_WAIT_WH_READY:
            IF PDraggUnload_WH_SubStates = PDraggUnload_WH_SubState.PUNLDRG_U17_DONE THEN
              PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U32_DONE;
            END_IF;

          PDraggUnload_SH_SubState.PUNLDRG_U32_DONE:
            Local_Shuttle_State := Shuttle_States.U_Punch_positioning_in_WH_progress;
            PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U34_INIT;

          PDraggUnload_SH_SubState.PUNLDRG_U34_INIT:
            SH_DR_MoveRequested := FALSE;
            SH_DR_Done := FALSE;
            PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U34_PIN2_INSIDE_CMD;

          PDraggUnload_SH_SubState.PUNLDRG_U34_PIN2_INSIDE_CMD:
            Fmsc_PLC_IO.Shuttle_Pin2_Move_Outside_Valve := FALSE;
            Fmsc_PLC_IO.Shuttle_Pin2_Move_Inside_Valve := TRUE;
            PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U34_WAIT_PIN2_INSIDE;

          PDraggUnload_SH_SubState.PUNLDRG_U34_WAIT_PIN2_INSIDE:
            IF Fmsc_PLC_IO.Shuttle_Pin2_Inside THEN
              Fmsc_PLC_IO.Shuttle_Pin2_Move_Inside_Valve := FALSE;
              PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U34_PIN1_INSIDE_CMD;
            END_IF;

          PDraggUnload_SH_SubState.PUNLDRG_U34_PIN1_INSIDE_CMD:
            Fmsc_PLC_IO.Shuttle_Pin1_Move_Outside_Valve := FALSE;
            Fmsc_PLC_IO.Shuttle_Pin1_Move_Inside_Valve := TRUE;
            PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U34_WAIT_PIN1_INSIDE;

          PDraggUnload_SH_SubState.PUNLDRG_U34_WAIT_PIN1_INSIDE:
            IF Fmsc_PLC_IO.Shuttle_Pin1_Inside THEN
              Fmsc_PLC_IO.Shuttle_Pin1_Move_Inside_Valve := FALSE;
              PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U34_DONE;
            END_IF;

          PDraggUnload_SH_SubState.PUNLDRG_U34_DONE:
            PunchUsed[TO_INT(Retain_ATCDragPunchList[Current_PunchIndex,2]), TO_INT(Retain_ATCDragPunchList[Current_PunchIndex,3])] := FALSE;
            PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U34_NEXT_PUNCH_INIT;

          PDraggUnload_SH_SubState.PUNLDRG_U34_NEXT_PUNCH_INIT:
            tmpInt2 := 0;
            IF Retain_DragPunchList[Current_PunchIndex,12] <> '' THEN
              tmpInt2 := STRING_TO_INT(Retain_DragPunchList[Current_PunchIndex,12]);
            END_IF;

            IF (tmpInt2 >= 1) AND (tmpInt2 <= 15) THEN
              FOR k := 1 TO 12 DO
                PFields_Old[tmpInt2, k] := '';
              END_FOR;
              OldAssignedDrawer[tmpInt2] := 0;
              OldAssignedSlot[tmpInt2] := 0;
            END_IF;

            Current_PunchIndex := Current_PunchIndex + 1;
            IF Current_PunchIndex > DragPunchCount THEN
              Current_PunchIndex := 0;
              Current_PunchLoaded := FALSE;
              UnloadDone := TRUE;
              initiated := FALSE;
              UnloadBusy := FALSE;
              DragPrepared := FALSE;

              Local_WareHouse_State := 0;
              Local_Shuttle_State := 0;
              Local_RotArm_State := 0;

              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_IDLE;
              PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.SH_IDLE;
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_IDLE;
            ELSE
              Current_PunchLoaded := FALSE;
              LastInitializedPunch := 0;

              WH_TargetDrawerIdx := 0;
              WH_TargetRackID := 0;
              WH_NeedRackBring := FALSE;
              WH_NeedRackChange := FALSE;
              WH_SameRack := FALSE;

              WH_R_MoveRequested := FALSE;
              WH_X_MoveRequested := FALSE;
              WH_R_Done := FALSE;
              WH_X_Done := FALSE;
              WH_WaitRackInPos := FALSE;

              WH_R_LoadRequested := FALSE;
              WH_R_LoadDone := FALSE;
              R_Not_Safety_Zone := FALSE;
              Rack_Presence_Error := FALSE;

              SH_DX_MoveRequested := FALSE;
              SH_DX_Done := FALSE;
              SH_DR_MoveRequested := FALSE;
              SH_DR_Done := FALSE;
              SH_Z_MoveRequested := FALSE;
              SH_Z_Done := FALSE;

              RotRequired := FALSE;
              RotArm_RotationDone := FALSE;
              RotArm_PunchReleased := FALSE;

              Local_WareHouse_State := WareHouse_States.PRG_Proccesing;
              PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_INIT;

              Local_Shuttle_State := Shuttle_States.PRG_Proccesing;
              PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U26_INIT;
            END_IF;

          ELSE
            PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U26_INIT;
        END_CASE;

      ELSE
        ;
    END_CASE;
  END_IF;
