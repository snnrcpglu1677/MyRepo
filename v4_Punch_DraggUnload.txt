FUNCTION_BLOCK Punch_DraggUnload

VAR_INPUT
  UnloadStartCmd      : BOOL;    (* State_Machine'den gelen başlat komutu *)
  Reset               : BOOL;    (* SOFT reset: state/substate resetlemez, sadece requestleri keser *)
  Abort_Cycle         : BOOL;    (* HARD abort: tüm FB içi state/substate/flagleri resetler *)
  Automatic_Page      : BOOL;    (* TRUE: otomatik sayfa aktif, FALSE: otomatik dışı -> pause zorunlu *)
END_VAR

VAR_IN_OUT
  General_Stop        : BOOL;
  General_Start       : BOOL;    (* Enable/pause: FALSE iken pause, TRUE iken devam *)
END_VAR

VAR_OUTPUT
  UnloadStartAck      : BOOL;    (* State_Machine'e ACK *)
  UnloadBusy          : BOOL;    (* FB aktif çalışıyor *)
  UnloadError         : BOOL;    (* hata flag *)
  UnloadDone          : BOOL;    (* tüm unload bitti *)
END_VAR

VAR RETAIN
  {attribute 'ac_persist':='PersistenceChannel'} Local_Shuttle_State     : INT := 0;
  {attribute 'ac_persist':='PersistenceChannel'} Local_WareHouse_State   : INT := 0;
  {attribute 'ac_persist':='PersistenceChannel'} Local_RotArm_State      : INT := 0;

  {attribute 'ac_persist':='PersistenceChannel'} UnloadPunchCount        : INT := 0;
  {attribute 'ac_persist':='PersistenceChannel'} DragPunchCount          : INT := 0;

  {attribute 'ac_persist':='PersistenceChannel'} PDraggUnload_WH_SubStates     : INT := 0;
  {attribute 'ac_persist':='PersistenceChannel'} PDraggUnload_SH_SubStates     : INT := 0;
  {attribute 'ac_persist':='PersistenceChannel'} PDraggUnload_RotArm_SubStates : INT := 0;

  lastStartCmd          : BOOL := FALSE;
  initiated             : BOOL := FALSE;
  DragPrepared          : BOOL := FALSE;

  Current_PunchIndex    : INT := 0;
  Current_PunchLoaded   : BOOL := FALSE;
  LastInitializedPunch  : INT := 0;

  (* Drag ordering helper arrays *)
  DragSrcIndex          : ARRAY[1..15] OF INT;
  DragDrawerNo          : ARRAY[1..15] OF INT;
  DragSlotNo            : ARRAY[1..15] OF INT;
  DragOrderUsed         : ARRAY[1..15] OF BOOL;

  (* Warehouse helper flags *)
  WH_TargetDrawerIdx    : INT := 0;
  WH_TargetRackID       : INT := 0;
  WH_NeedRackBring      : BOOL := FALSE;
  WH_NeedRackChange     : BOOL := FALSE;
  WH_SameRack           : BOOL := FALSE;

  WH_R_MoveRequested    : BOOL := FALSE;
  WH_X_MoveRequested    : BOOL := FALSE;
  WH_R_Done             : BOOL := FALSE;
  WH_X_Done             : BOOL := FALSE;
  WH_WaitRackInPos      : BOOL := FALSE;

  WH_Eps                : REAL := 0.01;

  WH_R_LoadRequested    : BOOL := FALSE;
  WH_R_LoadDone         : BOOL := FALSE;
  R_Not_Safety_Zone     : BOOL := FALSE;
  Rack_Presence_Error   : BOOL := FALSE;

  (* Shuttle helper flags *)
  SH_DX_MoveRequested   : BOOL := FALSE;
  SH_DX_Done            : BOOL := FALSE;
  SH_DR_MoveRequested   : BOOL := FALSE;
  SH_DR_Done            : BOOL := FALSE;
  SH_Z_MoveRequested    : BOOL := FALSE;
  SH_Z_Done             : BOOL := FALSE;

  (* Rotation arm helper flags *)
  RA_W_MoveRequested    : BOOL := FALSE;
  RA_W_Done             : BOOL := FALSE;
  RotArm_ReadyForUnload : BOOL := FALSE;
  RotArm_RotationDone   : BOOL := FALSE;
  RotArm_PunchReleased  : BOOL := FALSE;
  RotRequired           : BOOL := FALSE;
END_VAR

VAR
  risingStart           : BOOL;
  AutoRunEnable         : BOOL;
  i                     : INT;
  j                     : INT;
  k                     : INT;
  tmpInt                : INT;
  tmpInt2               : INT;
  tmpSrc                : INT;

  ClampOff_Timer        : TON;
  ClampOn_Timer         : TON;
END_VAR


(* ------------------------------------------------------------ *)
(* Combined enable: must be on Auto page AND General_Start TRUE *)
(* ------------------------------------------------------------ *)
AutoRunEnable := (General_Start AND Automatic_Page);
ClampOff_Timer(PT := TO_TIME(TO_INT(Retain_GeneralParameters[4] * 1000)));
ClampOn_Timer(PT := TO_TIME(TO_INT(Retain_GeneralParameters[3] * 1000)));

risingStart := (UnloadStartCmd AND NOT lastStartCmd);


(* ------------------------------------------------------------ *)
(* HARD ABORT: reset all FB-internal state (lists are untouched)*)
(* ------------------------------------------------------------ *)
IF Abort_Cycle THEN
  UnloadStartAck := FALSE;
  UnloadBusy := FALSE;
  UnloadError := FALSE;
  UnloadDone := FALSE;

  lastStartCmd := FALSE;
  initiated := FALSE;
  DragPrepared := FALSE;

  Current_PunchIndex := 0;
  Current_PunchLoaded := FALSE;
  LastInitializedPunch := 0;

  UnloadPunchCount := 0;
  DragPunchCount := 0;

  Local_Shuttle_State := 0;
  Local_WareHouse_State := 0;
  Local_RotArm_State := 0;

  PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_IDLE;
  PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.SH_IDLE;
  PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_IDLE;

  WH_TargetDrawerIdx := 0;
  WH_TargetRackID := 0;
  WH_NeedRackBring := FALSE;
  WH_NeedRackChange := FALSE;
  WH_SameRack := FALSE;

  WH_R_MoveRequested := FALSE;
  WH_X_MoveRequested := FALSE;
  WH_R_Done := FALSE;
  WH_X_Done := FALSE;
  WH_WaitRackInPos := FALSE;

  WH_R_LoadRequested := FALSE;
  WH_R_LoadDone := FALSE;
  R_Not_Safety_Zone := FALSE;
  Rack_Presence_Error := FALSE;

  SH_DX_MoveRequested := FALSE;
  SH_DX_Done := FALSE;
  SH_DR_MoveRequested := FALSE;
  SH_DR_Done := FALSE;
  SH_Z_MoveRequested := FALSE;
  SH_Z_Done := FALSE;

  RA_W_MoveRequested := FALSE;
  RA_W_Done := FALSE;
  RotArm_ReadyForUnload := FALSE;
  RotArm_RotationDone := FALSE;
  RotArm_PunchReleased := FALSE;
  RotRequired := FALSE;

  arAxes[EN_Axes.ATC_DR].AxisAutoSetVelocity := Retain_DRParameters[1];

  arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_DX].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_DR].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_Z].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := FALSE;

  RETURN;
END_IF


(* ------------------------------------------------------------ *)
(* PAUSE: freeze states, cut requests                           *)
(* ------------------------------------------------------------ *)
IF NOT AutoRunEnable AND NOT PLC_PRG.HMI.Drawer_Demand_Enable THEN
  arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_DX].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_DR].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_Z].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := FALSE;

  lastStartCmd := UnloadStartCmd;
  RETURN;
END_IF


(* ------------------------------------------------------------ *)
(* SOFT RESET: do not reset states/substates; only cut requests *)
(* ------------------------------------------------------------ *)
IF Reset THEN
  arAxes[EN_Axes.ATC_R].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_X].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_DX].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_DR].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_Z].AxisTravelRequestAbs := FALSE;
  arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := FALSE;
END_IF


(* ------------------------------------------------------------ *)
(* START + DRAG LIST PREPARATION                                *)
(* ------------------------------------------------------------ *)
IF risingStart THEN
  initiated := TRUE;
  UnloadBusy := TRUE;
  UnloadError := FALSE;
  UnloadDone := FALSE;
  DragPrepared := FALSE;

  UnloadPunchCount := 0;
  DragPunchCount := 0;

  FOR i := 1 TO 15 DO
    DragSrcIndex[i] := 0;
    DragDrawerNo[i] := 0;
    DragSlotNo[i] := 0;
    DragOrderUsed[i] := FALSE;
  END_FOR;

  FOR i := 1 TO 15 DO
    IF (Retain_UnloadPunchList[i,3] <> '') THEN
      UnloadPunchCount := UnloadPunchCount + 1;
    END_IF;
  END_FOR;

  FOR i := 1 TO 15 DO
    IF (Retain_UnloadPunchList[i,3] <> '') AND (Retain_UnloadPunchList[i,2] = '1') THEN
      DragPunchCount := DragPunchCount + 1;
      DragSrcIndex[DragPunchCount] := i;

      IF Retain_UnloadPunchList[i,10] <> '' THEN
        DragDrawerNo[DragPunchCount] := STRING_TO_INT(Retain_UnloadPunchList[i,10]);
      ELSE
        DragDrawerNo[DragPunchCount] := 0;
      END_IF;

      IF Retain_UnloadPunchList[i,11] <> '' THEN
        DragSlotNo[DragPunchCount] := STRING_TO_INT(Retain_UnloadPunchList[i,11]);
      ELSE
        DragSlotNo[DragPunchCount] := 0;
      END_IF;
    END_IF;
  END_FOR;

  FOR i := 1 TO DragPunchCount - 1 DO
    FOR j := i + 1 TO DragPunchCount DO
      IF (DragDrawerNo[j] < DragDrawerNo[i]) OR
         ((DragDrawerNo[j] = DragDrawerNo[i]) AND (DragSlotNo[j] < DragSlotNo[i])) THEN

        tmpInt := DragSrcIndex[i];
        DragSrcIndex[i] := DragSrcIndex[j];
        DragSrcIndex[j] := tmpInt;

        tmpInt := DragDrawerNo[i];
        DragDrawerNo[i] := DragDrawerNo[j];
        DragDrawerNo[j] := tmpInt;

        tmpInt := DragSlotNo[i];
        DragSlotNo[i] := DragSlotNo[j];
        DragSlotNo[j] := tmpInt;
      END_IF;
    END_FOR;
  END_FOR;

  FOR i := 1 TO DragPunchCount DO
    tmpSrc := DragSrcIndex[i];

    FOR k := 1 TO 12 DO
      Retain_DragPunchList[i,k] := Retain_UnloadPunchList[tmpSrc,k];
      Retain_ATCDragPunchList[i,k] := Retain_ATCUnloadPunchList[tmpSrc,k];
    END_FOR;
  END_FOR;

  IF DragPunchCount = 0 THEN
    Current_PunchIndex := 0;
    Current_PunchLoaded := FALSE;
    UnloadDone := TRUE;
    initiated := FALSE;

    Local_WareHouse_State := 0;
    Local_Shuttle_State := 0;
    Local_RotArm_State := 0;

    PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_IDLE;
    PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.SH_IDLE;
    PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_IDLE;
  ELSE
    Current_PunchIndex := 1;
    Current_PunchLoaded := FALSE;
    UnloadDone := FALSE;
    DragPrepared := TRUE;

    Local_Shuttle_State := Shuttle_States.PRG_Proccesing;
    Local_WareHouse_State := WareHouse_States.PRG_Proccesing;
    Local_RotArm_State := RotationArm_States.U_RotArm_Init_And_Home;

    PDraggUnload_SH_SubStates := PDraggUnload_SH_SubState.PUNLDRG_U26_INIT;
    PDraggUnload_WH_SubStates := PDraggUnload_WH_SubState.PUNLDRG_U14_INIT;
    PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_CHECK_POSITION;
  END_IF;

  UnloadStartAck := TRUE;

ELSIF initiated AND UnloadStartCmd THEN
  UnloadStartAck := TRUE;
  UnloadBusy := TRUE;

ELSIF initiated AND NOT UnloadStartCmd THEN
  UnloadStartAck := FALSE;
  UnloadBusy := TRUE;

ELSE
  UnloadStartAck := FALSE;
  IF NOT initiated THEN
    UnloadBusy := FALSE;
  END_IF;
END_IF


(* ------------------------------------------------------------ *)
(* ENGINE FLOWS                                                 *)
(* ------------------------------------------------------------ *)
IF initiated AND DragPrepared THEN

  (* Rotation Arm engine *)
  IF UnloadStartCmd THEN
    CASE Local_RotArm_State OF
      RotationArm_States.U_RotArm_Init_And_Home:
        CASE PDraggUnload_RotArm_SubStates OF
          PDraggUnload_RotArm_SubState.ROT_IDLE:
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_CHECK_POSITION;

          PDraggUnload_RotArm_SubState.ROT_INIT_CHECK_POSITION:
            RotArm_ReadyForUnload := FALSE;
            RotArm_RotationDone := FALSE;
            RotArm_PunchReleased := FALSE;
            RA_W_MoveRequested := FALSE;
            RA_W_Done := FALSE;

            IF Fmsc_PLC_IO.Rotation_Arm_Inside THEN
              IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - 90.0) <= WH_Eps THEN
                PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_MOVE_OUTSIDE_CMD;
              ELSE
                PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_W_TO_90_REQ;
              END_IF;
            ELSIF Fmsc_PLC_IO.Rotation_Arm_Outside THEN
              IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - 180.0) <= WH_Eps THEN
                PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_LOCK_CMD;
              ELSE
                PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_W_TO_180_REQ;
              END_IF;
            ELSE
              UnloadError := TRUE;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INIT_W_TO_90_REQ:
            arAxes[EN_Axes.ATC_W].TargetPos := 90.0;
            arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := TRUE;
            RA_W_MoveRequested := TRUE;
            RA_W_Done := FALSE;
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_W_TO_90_WAIT;

          PDraggUnload_RotArm_SubState.ROT_INIT_W_TO_90_WAIT:
            IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - arAxes[EN_Axes.ATC_W].TargetPos) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := FALSE;
              RA_W_MoveRequested := FALSE;
              RA_W_Done := TRUE;
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_MOVE_OUTSIDE_CMD;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INIT_MOVE_OUTSIDE_CMD:
            Fmsc_PLC_IO.Rotation_Arm_Move_Inside_Valve := FALSE;
            Fmsc_PLC_IO.Rotation_Arm_Move_Outside_Valve := TRUE;
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_WAIT_OUTSIDE;

          PDraggUnload_RotArm_SubState.ROT_INIT_WAIT_OUTSIDE:
            IF Fmsc_PLC_IO.Rotation_Arm_Outside THEN
              Fmsc_PLC_IO.Rotation_Arm_Move_Outside_Valve := FALSE;
              IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - 180.0) <= WH_Eps THEN
                PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_LOCK_CMD;
              ELSE
                PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_W_TO_180_REQ;
              END_IF;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INIT_W_TO_180_REQ:
            arAxes[EN_Axes.ATC_W].TargetPos := 180.0;
            arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := TRUE;
            RA_W_MoveRequested := TRUE;
            RA_W_Done := FALSE;
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_W_TO_180_WAIT;

          PDraggUnload_RotArm_SubState.ROT_INIT_W_TO_180_WAIT:
            IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - arAxes[EN_Axes.ATC_W].TargetPos) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := FALSE;
              RA_W_MoveRequested := FALSE;
              RA_W_Done := TRUE;
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_LOCK_CMD;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INIT_LOCK_CMD:
            Fmsc_PLC_IO.Rotation_Arm_Unlocking_Valve := FALSE;
            Fmsc_PLC_IO.Rotation_Arm_Locking_Valve := TRUE;
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_WAIT_LOCKED;

          PDraggUnload_RotArm_SubState.ROT_INIT_WAIT_LOCKED:
            IF Fmsc_PLC_IO.Rotation_Arm_Locked_CH2 AND Fmsc_PLC_IO.Rotation_Arm_Locked_CH3 THEN
              RotArm_ReadyForUnload := TRUE;
              Local_RotArm_State := RotationArm_States.U_RotArm_Ready_For_Unload;
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_READY;
            END_IF;

        ELSE
          PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INIT_CHECK_POSITION;
        END_CASE;

      RotationArm_States.U_RotArm_Ready_For_Unload:
        CASE PDraggUnload_RotArm_SubStates OF
          PDraggUnload_RotArm_SubState.ROT_READY:
            RotArm_ReadyForUnload := TRUE;
            IF RotRequired THEN
              Local_RotArm_State := RotationArm_States.U_RotArm_Rotation_Interrupt_Active;
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_WAIT_PUNCH_ON_ARM;
            END_IF;
        ELSE
          PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_READY;
        END_CASE;

      RotationArm_States.U_RotArm_Rotation_Interrupt_Active:
        CASE PDraggUnload_RotArm_SubStates OF
          PDraggUnload_RotArm_SubState.ROT_INT_WAIT_PUNCH_ON_ARM:
            RotArm_ReadyForUnload := FALSE;
            RotArm_RotationDone := FALSE;
            RotArm_PunchReleased := FALSE;
            IF Fmsc_PLC_IO.Punch_Locked_on_Rotation_Arm THEN
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_WAIT_SHUTTLE_CLEAR;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INT_WAIT_SHUTTLE_CLEAR:
            IF (PDraggUnload_SH_SubStates = PDraggUnload_SH_SubState.PUNLDRG_U28_DONE) THEN
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_ARM_UNLOCK_CMD;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INT_ARM_UNLOCK_CMD:
            Fmsc_PLC_IO.Rotation_Arm_Unlocking_Valve := TRUE;
            Fmsc_PLC_IO.Rotation_Arm_Locking_Valve := FALSE;
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_WAIT_ARM_UNLOCKED;

          PDraggUnload_RotArm_SubState.ROT_INT_WAIT_ARM_UNLOCKED:
            IF Fmsc_PLC_IO.Rotation_Arm_Unlocked_CH2 AND Fmsc_PLC_IO.Rotation_Arm_Unlocked_CH3 THEN
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_W_TO_ROTATED_REQ;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INT_W_TO_ROTATED_REQ:
            arAxes[EN_Axes.ATC_W].TargetPos := 0.0;
            arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := TRUE;
            RA_W_MoveRequested := TRUE;
            RA_W_Done := FALSE;
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_W_TO_ROTATED_WAIT;

          PDraggUnload_RotArm_SubState.ROT_INT_W_TO_ROTATED_WAIT:
            IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - arAxes[EN_Axes.ATC_W].TargetPos) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := FALSE;
              RA_W_MoveRequested := FALSE;
              RA_W_Done := TRUE;
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_ARM_LOCK_CMD;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INT_ARM_LOCK_CMD:
            Fmsc_PLC_IO.Rotation_Arm_Locking_Valve := TRUE;
            Fmsc_PLC_IO.Rotation_Arm_Unlocking_Valve := FALSE;
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_WAIT_ARM_LOCKED;

          PDraggUnload_RotArm_SubState.ROT_INT_WAIT_ARM_LOCKED:
            IF Fmsc_PLC_IO.Rotation_Arm_Locked_CH2 AND Fmsc_PLC_IO.Rotation_Arm_Locked_CH3 THEN
              RotArm_RotationDone := TRUE;
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_WAIT_SHUTTLE_PICKBACK;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INT_WAIT_SHUTTLE_PICKBACK:
            IF (PDraggUnload_SH_SubStates = PDraggUnload_SH_SubState.PUNLDRG_U30_WAIT_ROTARM_RELEASED) THEN
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_UNLOCK_CMD;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INT_UNLOCK_CMD:
            Fmsc_PLC_IO.Rotation_Arm_Unlocking_Valve := TRUE;
            Fmsc_PLC_IO.Rotation_Arm_Locking_Valve := FALSE;
            Fmsc_PLC_IO.Rotation_Arm_Punch_Lock_Valve := FALSE;
            Fmsc_PLC_IO.Rotation_Arm_Punch_Unlock_Valve := TRUE;
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_WAIT_PUNCH_RELEASED;

          PDraggUnload_RotArm_SubState.ROT_INT_WAIT_PUNCH_RELEASED:
            IF NOT Fmsc_PLC_IO.Punch_Locked_on_Rotation_Arm THEN
              RotArm_PunchReleased := TRUE;
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_RETURN_UNLOCK_CMD;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INT_RETURN_UNLOCK_CMD:
            Fmsc_PLC_IO.Rotation_Arm_Unlocking_Valve := TRUE;
            Fmsc_PLC_IO.Rotation_Arm_Locking_Valve := FALSE;
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_RETURN_WAIT_UNLOCKED;

          PDraggUnload_RotArm_SubState.ROT_INT_RETURN_WAIT_UNLOCKED:
            IF Fmsc_PLC_IO.Rotation_Arm_Unlocked_CH2 AND Fmsc_PLC_IO.Rotation_Arm_Unlocked_CH3 THEN
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_RETURN_W_TO_180_REQ;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INT_RETURN_W_TO_180_REQ:
            IF arAxes[EN_Axes.ATC_Z].ActualPos <= TO_LREAL(Retain_ZParameters[8]) THEN
              arAxes[EN_Axes.ATC_W].TargetPos := 180.0;
              arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := TRUE;
              RA_W_MoveRequested := TRUE;
              RA_W_Done := FALSE;
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_RETURN_W_TO_180_WAIT;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INT_RETURN_W_TO_180_WAIT:
            IF ABS(arAxes[EN_Axes.ATC_W].ActualPos - arAxes[EN_Axes.ATC_W].TargetPos) <= WH_Eps THEN
              arAxes[EN_Axes.ATC_W].AxisTravelRequestAbs := FALSE;
              RA_W_MoveRequested := FALSE;
              RA_W_Done := TRUE;
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_RETURN_LOCK_CMD;
            END_IF;

          PDraggUnload_RotArm_SubState.ROT_INT_RETURN_LOCK_CMD:
            Fmsc_PLC_IO.Rotation_Arm_Locking_Valve := TRUE;
            Fmsc_PLC_IO.Rotation_Arm_Unlocking_Valve := FALSE;
            PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_RETURN_WAIT_LOCKED;

          PDraggUnload_RotArm_SubState.ROT_INT_RETURN_WAIT_LOCKED:
            IF Fmsc_PLC_IO.Rotation_Arm_Locked_CH2 AND Fmsc_PLC_IO.Rotation_Arm_Locked_CH3 THEN
              PDraggUnload_RotArm_SubStates := PDraggUnload_RotArm_SubState.ROT_INT_DONE;
            END_IF;
